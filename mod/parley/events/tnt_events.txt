########################################################################
# PARLEY: THE NEGOTIATING TABLE — the confirmation event
#
# Contract 10-CONTRACT.md §6.4.
#
# WHY AN EVENT AT ALL. A scripted_gui `effect` runs the instant the button
# is clicked, with no confirmation dialog of any kind. Vanilla routes
# irreversible scripted_gui actions through trigger_event precisely for
# this reason — GAME\common\scripted_guis\00_religion.txt:64-67 carries
# PDX's own comment `#Are you sure?`. Ceding titles and swearing fealty
# must never happen on a single misclick.
#
# HOW IT IS REACHED. The tnt_send_offer scripted_gui does
#   effect = {
#       var:tnt_partner ?= { save_scope_as = tnt_deal_partner }
#       trigger_event = tnt_diplomacy.0001
#   }
# so scope:tnt_deal_partner arrives with the event. trigger_event fires on
# the scope it is called in, which is the player, so there is no multiplayer
# scoping problem: only the player who clicked sees this window.
#
# *** THE SCOPE IS NAMED tnt_deal_partner, NOT tnt_p — DEFECT D4 ***
# (30-V2-DESIGN §9 D4.) The two option triggers below evaluate
# tnt_ai_accept_value, and every value in that chain opens with contract
# R0.4's prologue `var:tnt_partner ?= { save_temporary_scope_as = tnt_p }`.
# Against v1's permanently-saved `tnt_p` that produced, at THIS file's
# lines 84 and 98:
#   "[jomini_eventscope.cpp:565]: Trying to add the temporary target 'tnt_p'
#    which has the same name as a permanent target"
# and the values silently read the permanent scope instead of their own.
# The permanent scope was renamed; the value files keep `tnt_p`. The loc key
# tnt_confirm_desc prints it as [tnt_deal_partner.GetShortUIName] and must
# be renamed in BOTH .yml files with this file.
#
# THE THREE OPTIONS.
#   1. Send  — visible only when tnt_ai_accept_value > 0. Its tooltip is
#      auto-generated by running tnt_apply_deal_effect in tooltip mode, so
#      the player reads the complete, real consequence list before
#      committing. This is the same mechanism vanilla decisions use.
#      On a successful click this option, and ONLY this option, closes the
#      negotiation window afterwards (user item 3.5). A stale package fails
#      preflight, shows tnt_err_deal_changed and leaves the table intact.
#   2. Refused — visible only when tnt_ai_accept_value <= 0. The terms are
#      still previewed, via show_as_tooltip, which renders the effect's
#      tooltip WITHOUT running it
#      (GAME\common\decisions\00_lifestyle_decisions.txt:404-407, :463-466),
#      and the reason is stated with a custom_tooltip. Only
#      tnt_notify_rejected_effect actually runs; the offer stays on the
#      table so the player can sweeten it.
#   3. Think again — a pure cancel. No effect, no result banner, nothing
#      cleared. The window is still open behind the event.
#
# OPTION 2 IS REACHABLE ONLY BECAUSE tnt_send_offer's is_valid deliberately
# carries NO `tnt_ai_accept_value > 0` row. If that row is ever put back, the
# Propose button goes grey whenever the partner would refuse, this option can
# never fire, and tnt_notify_rejected_effect / tnt_result = 2 / the refused half
# of the in-window banner all become dead code. See the note in
# MOD\common\scripted_guis\tnt_20_scripted_guis.txt on tnt_send_offer.
#
# THE THRESHOLD IS FOLDED INTO tnt_ai_accept_value, so the only test
# anywhere in this mod is `> 0` / `<= 0`. Exactly 0 declines — proven twice
# in vanilla (GAME\localization\english\interactions_l_english.yml:23 and
# GAME\gui\interaction_templates.gui:3089).
#
# Comparing a NAMED SCRIPT VALUE against a literal is legal and standard
# (interaction_relative_hostage_value > -10,
# GAME\common\character_interactions\00_war.txt:1431). Comparing two
# variables is not, and never happens here.
########################################################################

namespace = tnt_diplomacy

tnt_diplomacy.0001 = {
	type = character_event
	title = tnt_confirm_title
	desc = tnt_confirm_desc
	theme = diplomacy

	# Belt and braces: a saved scope outlives its character, and the player
	# can be several frames away from having clicked Propose.
	trigger = {
		exists = scope:tnt_deal_partner
		exists = var:tnt_open
		exists = var:tnt_partner
		var:tnt_partner ?= { this = scope:tnt_deal_partner }
	}

	left_portrait = {
		character = root
		animation = personality_honorable
	}
	right_portrait = {
		character = scope:tnt_deal_partner
		animation = idle
	}

	########################################################################
	# ACCEPTED — apply the treaty for real, then close the window.
	########################################################################
	option = {
		name = tnt_confirm_opt_yes
		# Option triggers are rebuilt every frame. The trigger_if is an ordered
		# evaluation boundary: when the table is gone, the deep score is never read.
		trigger = {
			trigger_if = {
				limit = {
					exists = var:tnt_open
					exists = var:tnt_partner
					exists = scope:tnt_deal_partner
					var:tnt_partner ?= { this = scope:tnt_deal_partner }
				}
				tnt_ai_accept_value > 0
			}
			trigger_else = { always = no }
		}

		hidden_effect = { remove_variable = tnt_confirm_pending }

		if = {
			limit = {
				tnt_deal_preflight_trigger = { PLAYER = root  PARTNER = scope:tnt_deal_partner }
				trigger_if = {
					limit = { tnt_interests_enabled_value > 0 }
					tnt_ai_accept_value > 0
				}
			}
			tnt_apply_deal_effect = yes

		#===================================================================
		# USER ITEM 3.5 — the window auto-closes once the deal is committed.
		# (30-V2-DESIGN §1 row 3.5 and §9 D-Extra.)
		#
		# THREE THINGS ABOUT THIS LINE, ALL LOAD-BEARING:
		#
		# 1. It is HERE, in the accept option, and NOT inside
		#    tnt_apply_deal_effect. That effect is also rendered in tooltip
		#    mode by the refusal option's show_as_tooltip below, and will
		#    later be run by the AI-offer path where no window is open.
		#    Putting the close inside it would tear the window down while
		#    the player is merely HOVERING the button.
		#
		# 2. It is wrapped in hidden_effect so the option's auto-generated
		#    tooltip does not list the window teardown as a consequence of
		#    the treaty. hidden_effect suppresses tooltip rendering — the
		#    same property this mod already relies on at
		#    tnt_32_apply.txt (tnt_exec_vassal_effect), quoting PDX's own
		#    comment from
		#    GAME\common\scripted_effects\00_interaction_effects.txt:2158.
		#
		# 3. It is the LAST statement of the successful branch. tnt_close_window_effect
		#    calls tnt_reset_offer_effect, which removes tnt_result — the
		#    variable tnt_notify_accepted_effect writes at the very end of
		#    tnt_apply_deal_effect. Running the close first would leave the
		#    window open with a stale banner; running it last discards a
		#    banner nobody will see, which is correct, because the
		#    out-of-window feedback is the toast that has already fired.
		#
		# THE REFUSAL OPTION MUST NOT DO THIS. See the note below it.
		#
		# Fallback if closing from an event option ever fights the event
		# window's own close animation (30-V2-DESIGN §1 row 3.5): drive the
		# existing tnt_close_window scripted_gui from the option instead —
		# same effect, different entry point.
		#===================================================================
			hidden_effect = {
				tnt_close_window_effect = yes
			}
		}
		else = { custom_tooltip = tnt_err_deal_changed  hidden_effect = { tnt_notify_stale_deal_effect = yes } }
	}

	########################################################################
	# REFUSED — they will not sign. Show what was on the table and why.
	########################################################################
	# The caption is its OWN key, not tnt_msg_rejected. tnt_msg_rejected is the
	# past-tense result headline ("The offer is refused") used by the toast in
	# tnt_notify_rejected_effect and by the in-window banner; as a button label it
	# would read as a statement rather than an action.
	option = {
		name = tnt_confirm_opt_refused
		trigger = {
			trigger_if = {
				limit = {
					exists = var:tnt_open
					exists = var:tnt_partner
					exists = scope:tnt_deal_partner
					var:tnt_partner ?= { this = scope:tnt_deal_partner }
				}
				tnt_ai_accept_value <= 0
			}
			trigger_else = { always = no }
		}

		hidden_effect = { remove_variable = tnt_confirm_pending }

		# Preview only. show_as_tooltip renders the tooltip of the effects
		# inside it and does NOT execute them.
		show_as_tooltip = {
			tnt_apply_deal_effect = yes
		}
		custom_tooltip = tnt_err_refuse

		tnt_notify_rejected_effect = yes

		#===================================================================
		# *** DO NOT ADD tnt_close_window_effect TO THIS OPTION. ***
		# (30-V2-DESIGN §9 D-Extra, resolving contradiction C-10.)
		#
		# Item 3.5's auto-close is for the ACCEPT path only. The refusal
		# path keeps the window open ON PURPOSE so the player can sweeten
		# the offer and try again — that "sweeten and retry" loop is the
		# whole reason build-report change M3 removed the acceptance test
		# from tnt_send_offer.is_valid. Close the window here and
		# tnt_notify_rejected_effect, tnt_result = 2 and the refused half of
		# the in-window banner all become dead code again, roughly 60 lines
		# of it, and the player is back to a button that just stays grey.
		#
		# Note also that this option must NOT clear the table: the terms
		# stay exactly as they were so the retry starts from the last offer.
		#===================================================================
	}

	########################################################################
	# CANCEL — go back to the table with everything intact.
	########################################################################
	option = {
		name = tnt_confirm_opt_no
		trigger = {
			trigger_if = {
				limit = {
					exists = var:tnt_open
					exists = var:tnt_partner
					exists = scope:tnt_deal_partner
					var:tnt_partner ?= { this = scope:tnt_deal_partner }
				}
				always = yes
			}
			trigger_else = { always = no }
		}
		hidden_effect = { remove_variable = tnt_confirm_pending }
	}

	########################################################################
	# STALE SESSION — close only this obsolete confirmation. Never reset a
	# newer table whose partner no longer matches the scope captured above.
	########################################################################
	option = {
		name = tnt_confirm_opt_stale
		trigger = {
			trigger_if = {
				limit = {
					exists = var:tnt_open
					exists = var:tnt_partner
					exists = scope:tnt_deal_partner
					var:tnt_partner ?= { this = scope:tnt_deal_partner }
				}
				always = no
			}
			trigger_else = { always = yes }
		}
		custom_tooltip = tnt_err_session_changed
		hidden_effect = {
			remove_variable = tnt_confirm_pending
			if = {
				limit = {
					exists = var:tnt_open
					NOT = { exists = var:tnt_partner }
				}
				tnt_close_window_effect = yes
			}
		}
	}
}
