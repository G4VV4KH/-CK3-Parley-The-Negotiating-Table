# T45  tnt_can_war_each_other_trigger = { A = <character>  B = <character> }
#
# Scope: any. COULD THESE TWO ACTUALLY GO TO WAR WITH EACH OTHER? PURE - no
# custom_description anywhere in it - so it is safe inside an effect's limit, on
# the send path and in tooltip mode. SYMMETRIC: the answer does not depend on
# which ruler is passed as $A$.
#
# ARGUMENT CONTRACT, AND IT IS NOT COSMETIC: $A$ MUST BE AN ABSOLUTE REFERENCE
# (`root`, `scope:whatever`). $B$ may be absolute OR a root-relative var: path,
# which is what the three manual-window callers actually pass
# (B = var:tnt_partner). The reason is the 85,778-error rule spelled out at
# tnt_42_people_gates.txt:933-948 and repeated at the same-realm branch below:
# $PARAM$ substitution is textual, so a var: path is re-resolved in whatever
# scope the line sits in, and this trigger contains one nested scope change
# (`top_liege = { ... }`) whose interior is a THIRD character. That interior
# names $A$ only.
#
# V12 USER ITEM 4. A sub-vassal offered the player a ten-year truce; the player
# cannot declare war on a sub-vassal at all, only on his liege. The user's rule:
# such terms may pass only between two vassals of one realm, or two independent
# rulers of different realms.
#
# THIS IS VANILLA'S OWN RULE, NOT AN INVENTION. The engine trigger that gates
# every war declaration is can_attack_in_hierarchy, and its ONE real script call
# site states it:
#   GAME\common\character_interactions\00_war.txt:31
#     scope:actor = { ... can_attack_in_hierarchy = scope:recipient ... }
# The engine's own sentence for it:
#   GAME\localization\english\triggers\character_triggers_l_english.yml:621
#     "Can attack X as he is independent, the liege, or under the same liege."
# So a ruler may attack (a) an independent ruler, (b) his own liege, (c) a ruler
# under the same liege. THE RELATION IS ONE-DIRECTIONAL: a vassal may attack a
# foreign independent king; that king may not attack the vassal.
#
# WE REQUIRE IT IN BOTH DIRECTIONS, and the reason is the instrument, not taste.
# Our pact is ONE MUTUAL term - add_truce_both_ways, tnt_32_apply.txt E18 -
# priced once. Where only one side can attack, half the promise is empty and the
# single price is wrong by construction, and the mod has no one-way promise to
# sell instead (the third-party non-aggression family is retired). Written out,
# mutual reachability is exactly the two shapes the user named: two independent
# rulers - different realms by definition, since an independent ruler IS his own
# top liege - or two vassals under one top liege who are not each other's liege.
#
# WHY NOT JUST CALL can_attack_in_hierarchy TWICE. It would be shorter and it is
# tempting. Two reasons against. First, it is an engine trigger whose contents we
# cannot read: if it folds in has_truce or sworn_peace, a signed truce would stop
# the pair from ever renegotiating, which T19's header explicitly rules must stay
# possible. Second, every structural row below is separately proven and separately
# readable, and together they say the user's rule in the user's own words. The
# engine trigger stays in this header as the EVIDENCE that the rule is vanilla's.
#
# NOT TESTED HERE, deliberately: alive, imprisoned, already at war, already under
# truce, hegemon. Each is a state, not a hierarchy, and each already has its own
# row in the caller that needs it (T19, T13, T18, the door's is_valid, item 26).
# Keeping this trigger to ONE question is what makes it reusable.
#
# MOD-LOAD SAFE WITH AND WITHOUT AGOT: every row is an engine trigger. No trait,
# culture, faith, government or foreign scripted trigger is named.
# =============================================================================
tnt_can_war_each_other_trigger = {
	trigger_if = {
		limit = {
			exists = $A$
			exists = $B$
		}
		NOT = { $A$ = { this = $B$ } }
		$A$ = {
			is_ruler = yes			# REAL: GAME\common\character_interactions\00_alliance.txt:1124, :1127
			# Neither may stand above or below the other at ANY depth: a liege
			# cannot declare war on his own vassal or sub-vassal. This is the row
			# T19 and T13 used to carry alone, kept because the same-realm branch
			# below would otherwise let a duke and his own count through - both
			# are non-independent and both share a top liege.
			NOR = {
				target_is_liege_or_above = $B$	# REAL: GAME\common\scripted_triggers\00_war_and_peace_triggers.txt:187
				target_is_vassal_or_below = $B$	# REAL: GAME\common\scripted_triggers\00_war_and_peace_triggers.txt:199
			}
			# A suzerain cannot fight his tributary, nor a tributary his suzerain.
			NOT = { is_tributary_of_suzerain_or_above = $B$ }	# REAL: GAME\common\character_interactions\00_war.txt:113
		}
		$B$ = {
			is_ruler = yes
			NOT = { is_tributary_of_suzerain_or_above = $A$ }	# REAL: GAME\common\scripted_triggers\00_war_and_peace_triggers.txt:926
		}
		OR = {
			# BOTH INDEPENDENT. Two independent rulers are in different realms by
			# definition - each is his own top liege - so no realm row is needed
			# or even expressible here.
			AND = {
				$A$ = { is_independent_ruler = yes }	# REAL: GAME\common\character_interactions\00_fp3_interactions.txt:1534, 00_grant_titles_interaction.txt:280, :1087
				$B$ = { is_independent_ruler = yes }
			}
			# BOTH VASSALS OF ONE REALM. `top_liege = { ... }` as a scope change is
			# vanilla's own form, GAME\common\character_interactions\00_war.txt:157;
			# asking it through target_is_vassal_or_below avoids dotting a variable
			# path, which this mod has no precedent for.
			#
			# *** THE NESTED BLOCK DEREFERENCES $A$, NEVER $B$, AND THAT ORDER IS
			# *** LOAD-BEARING - IT IS THE 85,778-ERROR RULE OF
			# *** tnt_42_people_gates.txt:933-948. $PARAM$ SUBSTITUTION IS TEXTUAL, so
			# *** a caller who passes a var: path has that path RE-RESOLVED IN
			# *** WHATEVER SCOPE THE LINE SITS IN. Inside `top_liege = { ... }` the
			# *** scope is a THIRD character - the top liege - who carries none of the
			# *** player's variables. Every one of this trigger's call sites passes an
			# *** ABSOLUTE $A$ (`root`, or `scope:actor` in the sandbox), while $B$ is
			# *** `var:tnt_partner` on the three manual-window paths
			# *** (tnt_20_scripted_guis.txt:581, :607, :1339) and `$OTHER$` on the show
			# *** gates. Written the other way round - `$A$ = { top_liege = {
			# *** target_is_vassal_or_below = $B$ } }` - a VASSAL PLAYER would read
			# *** tnt_partner off his own liege, get an unset scope, and lose the
			# *** in-realm truce row that this very branch exists to keep. The question
			# *** is symmetric, so asking it from $B$'s side costs nothing.
			# *** A NEW CALLER MUST KEEP $A$ ABSOLUTE.
			AND = {
				$A$ = { is_independent_ruler = no }	# REAL: GAME\common\character_interactions\00_war.txt:213, 00_war_and_peace_triggers.txt:943
				$B$ = {
					is_independent_ruler = no
					top_liege = { target_is_vassal_or_below = $A$ }
				}
			}
		}
	}
	trigger_else = { always = no }
}

##  WHY THESE ARE TRIGGERS AND NOT scripted_guis, AND WHY THE GUI READS A
##  VARIABLE. This is the R11 resolution and it is not negotiable.
##
##  A GUI `visible` cannot call a scripted_gui: Law R11 forbids GetScriptedGui
##  in a per-row visible or down, because it runs the full script trigger every
##  frame per instance and vanilla uses that shape at only ~20 sites in the whole
##  game (00-ARCHITECTURE.md:511-517). Twenty-eight rows would be twenty-eight
##  trigger evaluations per frame, several of them list scans.
##
##  So the flow is:
##     window opens
##       -> tnt_open_window_effect (E1) sets tnt_partner, tnt_advanced,
##          resets the offer, refills the seven picker lists and the two spouse

# T38 / T39  tnt_show_hook_p_trigger / _r_trigger = { OTHER = <character> }

# STRUCTURAL: the hook cannot be created — the ruling's "a hook when one already
# exists and cannot be added". can_add_hook is the engine's own answer and it
# covers all of it: an existing hook of that type, the 5-year strong-hook
# cooldown (GAME\common\defines\00_defines.txt:270), and the invalid-target
# cases.
#
# DIRECTION, and it is easy to get backwards. A hook offered on the LEFT (_p) is
# a hook the PLAYER hands over ABOUT HIMSELF, so the partner ends up HOLDING it.
# tnt_exec_hook_effect runs in the RECEIVER's scope with TARGET = the giver
# (tnt_32_apply.txt:724-737, called at :~130 with
#   SIDE = p  GIVER = root                  RECEIVER = the partner
#   SIDE = r  GIVER = the partner           RECEIVER = root ).
# So the _p gate asks the PARTNER whether he can add a hook on ROOT, and the _r
# gate asks ROOT whether he can add a hook on the PARTNER. Mirror of the exec,
# by construction.
#
# can_add_hook  REAL: GAME\common\scripted_effects\00_hook_effects.txt:15, :31, :46
#                     (the three add_hook_if_possible_* wrappers we call)
#                     GAME\common\casus_belli_types\00_claim.txt:565, :669, :736
#                     GAME\common\casus_belli_types\00_civil_war.txt:568
#               and the `target = scope:x` argument form is verbatim from
#                     00_claim.txt:565-568   can_add_hook = { type = favor_hook  target = scope:claimant }
#
# *** WAVE 3b COORDINATED EDIT (item 2.3, hook tiers). *** When tnt_hook_p/_r
# become a 0..3 tier index and the TYPE argument becomes tnt_deal_favor_hook /
# tnt_deal_strong_hook / tnt_deal_eternal_hook, the `type = favor_hook` below
# must become an OR over the three new types — the row must stay visible if ANY
# tier can still be granted. Do not leave it testing favor_hook alone.
# =============================================================================
tnt_show_hook_p_trigger = {
	trigger_if = {
		limit = { exists = $OTHER$ }
		$OTHER$ = {
			can_add_hook = {
				type = favor_hook
				# `prev`, NOT `root`. Same trap as the script values: this trigger is
				# reached from a .gui gate built by the TntGate macro, where there
				# is no root to speak of - the run logged 6x `Event target link
				# 'root' returned an unset scope` plus 6x `can_add_hook trigger
				# [ target character was null ]` from here, i.e. the row was decided
				# by an error rather than by the hook.
				# `prev` inside a scope transition is the character we came FROM,
				# which is exactly "me" however this trigger was entered.
				# REAL: 7 vanilla can_add_hook sites use target = prev, verbatim at
				# GAME\common\activities\pulse_actions\feast_pulse_actions_oltner.txt:341, :387
				# and GAME\common\decisions\10_ach_oath_decisions.txt:1372, :1391.
				target = prev
			}
		}
	}
	trigger_else = { always = no }
}

# =============================================================================
# T42  tnt_can_stage_pair_trigger = { P = <character>  R = <character> }
#
# SCOPE: THE PLAYER — it reads var:tnt_wed_list unqualified, and that list lives
# on the player (tnt_34_marriage.txt, THE STORAGE MODEL). Calling it in any other
# scope silently answers a different question.
#
# PURE — no custom_description anywhere, so it is safe in an effect's `limit` and
# in tooltip mode. Use T43 when a human needs to be told why.
#
# THE THREE REFUSALS, and they are exactly the three that tnt_wed_add reports:
#
#  1. NOT THE SAME PERSON. Both candidate lists can legitimately contain the same
#     character when the partner is the player's own close family, and vanilla's
#     could_marry_character_trigger does NOT catch it: its "Cannot marry self" row
#     is `NOT = { scope:can_marry_check = { is_spouse_of = $CHARACTER$ } }`
#     (00_marriage_triggers.txt:430-433), which is about spouses, not about one
#     character standing on both sides of the table. This row is ours and is the
#     same one tnt_marriage_sync_effect carries (tnt_34_marriage.txt:583).
#
#  2. VANILLA SAYS THEY MAY MARRY. can_marry_character_trigger
#     (00_marriage_triggers.txt:184-212) — gender and the same-sex game rule and
#     doctrine, recent divorce, consanguinity in BOTH faiths, polygamy slots,
#     concubine bookkeeping. Real call sites: 00_marriage_interactions.txt:55-57,
#     :172; 00_marriage_triggers.txt:415. NOT the could_ form: vanilla uses the
#     can_ form for both the marriage and the betrothal branch of
#     marriage_interaction_valid_target_trigger (:415, :426).
#     is_alive on both sides because the panel's candidate lists are a snapshot
#     taken at window open and a candidate can die while it sits there.
#
#  3. NOBODY STANDS AT TWO ALTARS. Four tests, not two, and all four are needed:
#     a character can already be in a committed pair on EITHER side, and neither
#     test finds the other.
#       * PLAYER-SIDE spouses ARE the list -> is_target_in_variable_list, paired
#         with has_variable_list exactly as vanilla pairs them
#         (00_character_interactions.txt:1060-1064).
#       * PARTNER-SIDE spouses are var:tnt_wed_partner on a list member, so they
#         can only be found by asking every member. `any_in_list` over a VARIABLE
#         list — REAL: GAME\common\script_values\00_scheme_values.txt:2290;
#         07_ep3_wars.txt:5868. `var:x ?= { this = <scope> }` is this mod's own
#         shipped read-back (tnt_70_on_actions.txt:174).
#     BOTH candidates are tested against BOTH sides — four rows — because the
#     panel can hand us a person who is already promised in the other direction.
#     This is the live equivalent of the tnt_wed_dup mirror
#     (tnt_34_marriage.txt:502-552); the mirror stays, because our own button
#     still needs it.
#
# WHAT IS DELIBERATELY NOT TESTED HERE:
#   * is_betrothed. It belongs to the CANDIDATE, not to the pair, and it is
#     already applied three times over — tnt_spouse_try_add_effect when the lists
#     are built, and the interaction's own can_be_picked and redirect.
#   * the Tier-4 game rule. It is an is_shown matter, and the interaction's
#     is_shown delegates it to T26.
#   * anything about the grand wedding. That is T44.
# =============================================================================
tnt_can_stage_pair_trigger = {
	trigger_if = {
		limit = {
			exists = $P$
			exists = $R$
		}

		NOT = { $P$ = { this = $R$ } }

		$P$ = {
			is_alive = yes
			can_marry_character_trigger = { CHARACTER = $R$ }
		}
		$R$ = { is_alive = yes }

		trigger_if = {
			limit = { has_variable_list = tnt_wed_list }
			NOR = {
				is_target_in_variable_list = {
					name = tnt_wed_list
					target = $P$
				}
				is_target_in_variable_list = {
					name = tnt_wed_list
					target = $R$
				}
				any_in_list = {
					variable = tnt_wed_list
					var:tnt_wed_partner ?= { this = $P$ }
				}
				any_in_list = {
					variable = tnt_wed_list
					var:tnt_wed_partner ?= { this = $R$ }
				}
			}
		}
	}
	trigger_else = { always = no }
}

		limit = {

			$B$ = { max_military_strength > 0 }
		}

	}
	trigger_else = { always = no }
}

tnt_grand_promised_trigger = {
	trigger_if = {
		limit = { exists = scope:secondary_actor }
		scope:grand_wedding_promise = yes
		# ONE PROMISE PER TREATY, and the preview must say so. Vanilla allows one
		# outstanding promise per host, and tnt_wed_stage_pair_effect enforces it
		# by ignoring the flag when var:tnt_wed_grand is already set. Without this
		# row a second couple would be previewed with a +40 the treaty is never
		# going to grant it.
		NOT = { exists = var:tnt_wed_grand }
	}
	trigger_else = {
		exists = var:tnt_wed_grand
		var:tnt_wed_grand > 0
		exists = var:tnt_wed_principal
		# `has_variable_list` FIRST. Without it error.log fills with
		# "variable_list_size trigger [Could not find list from read name flag
		# tnt_wed_list]" every time the balance is drawn before a couple exists -
		# the run of 2026-08-13 shows that line pointing at this trigger's caller.
		# The clauses of a trigger block are an unordered AND, so this is not a
		# short-circuit but a guard the engine honours: with the list absent the

		has_variable_list = tnt_wed_list
		variable_list_size = { name = tnt_wed_list  value > 0 }
	}
}

# THE PAIR HALF OF VANILLA'S GRAND-WEDDING GATE (00_marriage_interactions.txt:799-824):
# the celebrated spouse must be our own family AND must be the one whose house the
# children join. In the window that spouse is scope:secondary_actor; in the deal it
# is whichever couple was designated principal. The line is read through
# tnt_pair_is_matri_trigger in the window branch so that ticking "matrilineal" and
# ticking "grand wedding" interact exactly as vanilla makes them interact - promise
# a feast for the side that is marrying INTO the other house and vanilla refuses it.
### scope:tnt_me AND NOT root, in both branches. This trigger is called only from
### tnt_marriage_grand_value, which saves that scope on its first line precisely
### because an interaction's on_accept has no root - the 2026-08-13 run showed
### `Event target link 'root' returned an unset scope` coming through this path, and
### a family test against an unset scope is a silent NO, which is why the promise was
### never worth its 40 points.
tnt_grand_pair_half_trigger = {
	trigger_if = {
		limit = { exists = scope:secondary_actor }
		scope:secondary_actor ?= { is_close_or_extended_family_of = scope:tnt_me }
		scope:secondary_actor ?= {
			OR = {
				AND = {
					is_female = yes
					tnt_pair_is_matri_trigger = yes
				}
				AND = {
					is_male = yes
					NOT = { tnt_pair_is_matri_trigger = yes }
				}
			}
		}
	}
	trigger_else = {
		var:tnt_wed_principal ?= { is_close_or_extended_family_of = scope:tnt_me }
		var:tnt_wed_principal ?= {
			OR = {
				AND = {
					is_female = yes
					exists = var:tnt_wed_matri
				}
				AND = {
					is_male = yes
					NOT = { exists = var:tnt_wed_matri }
				}
			}
		}
	}
}

	trigger_if = {
		limit = {
			exists = $A$
			exists = $B$
		}
		$A$ = { NOT = { is_at_war_with = $B$ } }
	}
	trigger_else = { always = no }
}

		}
		$B$ = {

				target = $A$
				value >= -99
			}
		}
	}
	trigger_else = { always = no }
}

tnt_pair_opens_alliance_trigger = {
	trigger_if = {
		limit = {
			exists = $P$
			exists = $R$
			exists = $ME$
			exists = $OTHER$
		}
		# BOTH MUST ACTUALLY WED - see the header. Vanilla row shape:
		# 00_marriage_interactions.txt:122.
		$P$ = { is_adult = yes }
		$R$ = { is_adult = yes }
		OR = {
			# --- DOOR 1: the two rulers ARE the couple. vanilla :1186 -----
			AND = {
				$P$ = { this = $ME$ }
				$R$ = { this = $OTHER$ }
			}
			# --- DOOR 2: OTHER weds a close relative of ME. vanilla :1205-1219
			AND = {
				$R$ = { this = $OTHER$ }
				$P$ = {
					OR = {
						is_close_family_of = $ME$        # vanilla :1213
						is_nibling_of = $ME$             # vanilla :1214
						is_uncle_or_aunt_of = $ME$       # vanilla :1215
					}
				}
			}
			# --- DOOR 3: ME weds a close relative of OTHER. vanilla :1221-1235
			AND = {
				$P$ = { this = $ME$ }
				$R$ = {
					OR = {
						is_close_family_of = $OTHER$     # vanilla :1229
						is_nibling_of = $OTHER$          # vanilla :1230
						is_uncle_or_aunt_of = $OTHER$    # vanilla :1231
					}
				}
			}
			# --- DOOR 4: parent/child/sibling on BOTH sides. vanilla :1237-1258
			# The grandparent/grandchild half of close family is cut here by
			# vanilla itself - see the header.
			AND = {
				$P$ = {
					OR = {
						is_parent_of = $ME$              # vanilla :1245
						is_child_of = $ME$               # vanilla :1246
						is_sibling_of = $ME$             # vanilla :1247
					}
				}
				$R$ = {
					OR = {
						is_parent_of = $OTHER$           # vanilla :1251
						is_child_of = $OTHER$            # vanilla :1252
						is_sibling_of = $OTHER$          # vanilla :1253
					}
				}
			}
		}
	}
	trigger_else = { always = no }
}

# Note the tense. It is a question about a marriage THAT HAS NOT HAPPENED YET, and
# every shipped call site of ours must ask it BEFORE the bond is created. See the
# grant's own comment block in tnt_34_marriage.txt for the three reasons that
# position is load-bearing rather than merely tidy.
#
# (c) THE BINDING TABLE - the thing this wrapper exists to hold still.
#   evaluating scope   = an ARRANGER (a ruler at the table)
#   candidate          = the spouse-to-be on the EVALUATING ARRANGER's side
#   target             = THE OTHER ARRANGER  (never the other spouse)
#   target_candidate   = the other arranger's spouse-to-be
# All NINE vanilla sites in the whole install bind it that way. Measured this
# round by scanning every file under the game root for the identifier:
#   GAME\common\scripted_rules\00_rules.txt:807
#     - THE CANONICAL SITE, in scope:actor, under vanilla's comment at :806
#       "You're getting at least an alliance out of it":
#       candidate = scope:secondary_actor / target = scope:recipient /
#       target_candidate = scope:secondary_recipient.
#   GAME\common\scripted_triggers\00_marriage_triggers.txt:670
#     - same shape, in scope:actor.
#   GAME\common\scripted_effects\00_marriage_interaction_effects.txt:1662
#     - THE PROVEN CONTEXT FOR OUR USE: inside an EFFECT's `limit`. See (f).
#   GAME\common\scripted_modifiers\00_marriage_scripted_modifiers.txt:236, :314,
#     :350, :423, :484, :515
#     - six AI-accept sites, all in scope:recipient, all binding
#       candidate = scope:secondary_recipient / target = scope:actor /
#       target_candidate = scope:secondary_actor. THESE SIX ARE THE SAME BINDING
#       MIRRORED into the other arranger's scope, and that mirror is the proof
#       that the rule is "candidate = MY spouse, target = THE OTHER ARRANGER" and
#       not something positional. Whichever arranger asks, he names his own
#       candidate first and the man across the table second.
#
# (d) *** HOUSE LAW 10c IS ARMED THE MOMENT THIS TRIGGER EXISTS. ***
# *** yields_alliance IS A BLOCK-FORM, PARAMETER-BEARING ENGINE TRIGGER WITH NO
# *** common\trigger_localization\ ENTRY - MEASURED THIS ROUND, ZERO HITS OVER THE
# *** WHOLE FOLDER. THAT IS 10c'S EXACT SHIPPED-TWICE SHAPE: a parameterised block
# *** has nothing for the description walker to substitute, so if this trigger is
# *** ever reachable from a custom_description / custom_tooltip that BuildTooltip
# *** can walk, the player is shown `<key> has no localization` ON SCREEN and
# *** NOTHING IS WRITTEN TO ANY LOG. That is the defect that survived four reports
# *** and three fixes. EVERY SHIPPED CALL SITE OF T54 MUST SIT INSIDE A `limit`.
# *** A `limit` body is never described (house law 10b). No exceptions, ever.
#
# (e) PURE - no custom_description, no custom_tooltip, no text of any kind, exactly
# like T52 beside it. That purity is not decoration either: it is what lets a
# caller put the call inside a `limit` and know that nothing of ours reaches the
# walker.
#
# (f) WHERE IT IS PROVEN TO WORK, AND WHERE IT IS NOT.
#   PROVEN - inside an EFFECT's `limit`. That is vanilla's own shape at
#     00_marriage_interaction_effects.txt:1660-1667, where the betrothal is broken
#     first (:1658) and the predicate asked second (:1662) to decide whether an
#     alliance must be broken with it. Our grant is that same shape.
#   NOT PROVEN - inside a SCRIPT VALUE. `yields_alliance` occurs ZERO times in all
#     of GAME\common\script_values (measured this round). So the deal price and the
#     MCA candidate grade MAY NOT CALL T54: house law 1 wants a real vanilla call
#     site and there is none. Both must instead read a plain 0/1 that an EFFECT
#     wrote, which is exactly the shape tnt_34_marriage.txt already uses for
#     tnt_marriage_legal (its THREE MIRRORS block) and for the frozen
#     tnt_wed_gain/_loss/_net figures. If a later round wants the script-value
#     context, that is a MEASUREMENT to run in game, not an assumption to make.
#
# (g) EVERY ARGUMENT MUST BE AN ABSOLUTE HANDLE - `root`, `scope:x`, a saved scope.
# NEVER `this`, NEVER `prev`. Same rule and same reason as T51: these are text
# macros, the body is substituted verbatim, and a relative handle re-binds against
# whatever scope the substituted body happens to sit in. Here the cost of that
# mistake is not a wrong answer on screen - it is an alliance with the wrong man.
#
# THE WRAPPER IS DEFENSIVE ON PURPOSE. All four handles are tested with `exists`
# first, so a call made with a partner scope that was never saved answers `no`
# rather than throwing (R13). trigger_if / trigger_else is vanilla's own shape at
# 00_alliance.txt:23-28.
#
# EVERY IDENTIFIER, WITH A REAL VANILLA CALL SITE (house law 1). Paths relative to
# the vanilla game root.
#   yields_alliance             common\scripted_rules\00_rules.txt:807 (+8 more above)
#   exists                      everywhere
#   trigger_if / trigger_else   common\character_interactions\00_alliance.txt:23-28
# =============================================================================
tnt_pair_yields_alliance_trigger = {
	trigger_if = {
		limit = {
			exists = $P$
			exists = $R$
			exists = $ME$
			exists = $OTHER$
		}
		# The evaluating scope is an ARRANGER, so the binding is vanilla's
		# canonical one: my candidate, the man across the table, his candidate.
		# 00_rules.txt:807-811.
		$ME$ = {
			yields_alliance = {
				candidate = $P$
				target = $OTHER$
				target_candidate = $R$
			}
		}
	}
	trigger_else = { always = no }
}
