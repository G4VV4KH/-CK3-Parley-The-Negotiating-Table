	trigger_if = {
		limit = { exists = $CHARACTER$ }
		NOT = { this = $CHARACTER$ }
		# AGOT SUSPENDED (CK3 1.20): compatibility guard parked; awaiting the AGOT update.
		# V15, USER ITEM 3: THE WALL IS NOT A PARTY TO ANY NEGOTIATION, IN EITHER
		# ROLE. `this` is the ACTOR at all three of this trigger's call sites -
		# tnt_10_interaction.txt:414 inside `scope:actor = { }`,
		# tnt_37_ai_offer.txt with root = the AI proposer, and tnt_3b_log.txt's

		# never fires; the whole argument, the vanilla-only safety proof and the
		# cost table are in T48 below. Deliberately NOT folded into the NOR two
		# rows down: one statement per fact, and T48 is where the fact lives.
		# `PARAM = this` passed into a scripted trigger in a TRIGGER position is
		# vanilla's own shape - GAME\common\character_interactions\
		# 06_ep3_interactions.txt:33-36 and :48-51, both inside an iterator limit
		# (`any_valid_title_to_grant_trigger = { CANDIDATE = this ... }`).
		# What the substitution PRODUCES, `this = { <trigger block> }`, is REAL at
		# GAME\common\scripted_triggers\tgp_mandala_task_contract_triggers.txt
		# :108-110 and :123-125, and at GAME\common\character_interactions\
		# 00_diarch_interactions.txt:3016-3019.
		# TWO ADJACENT CALLS OF A trigger_if/trigger_else-SHAPED TRIGGER ARE TWO
		# INDEPENDENT CHAINS, not one - a `trigger_else` closes its own chain and a
		# following `trigger_if` opens a new one. REAL, five such pairs in a row
		# inside a single OR: GAME\common\scripted_triggers\00_activity_triggers.txt
		# :437-465 (petition_liege_has_main_skiller_trigger). Ours sit inside an
		# implicit AND, which is the easier case.
		# tnt_not_the_watch_trigger = { WHO = this }
		# tnt_not_the_watch_trigger = { WHO = $CHARACTER$ }
		$CHARACTER$ = {
			is_alive = yes				# REAL: GAME\common\character_interactions\00_choose_favorite_interaction.txt:103; also 00_courtier_and_guest_interactions.txt:172

# T2b  tnt_trade_prestige_trigger = { A = <character>  B = <character> }
# T2c  tnt_trade_piety_trigger    = { A = <character>  B = <character> }
#
# Scope: any. May $A$ and $B$ put PRESTIGE / PIETY on a negotiating table with
# each other, under the lobby rule for that currency?
#
# W2 TURNED TWO ON/OFF SWITCHES INTO TWO LADDERS AND THESE ARE THE ONLY PLACE
# THE LADDER IS SPELLED OUT. Nothing else in the mod may ask has_game_rule =
# tnt_trade_<cur>_* by hand any more: a hand-written `_on` test cannot see a
# middle rung, so it reads "off" for a rung that is in fact "on for this pair",
# and the mod would hide a row the lobby had allowed. The settings are declared
# in common\game_rules\tnt_80_game_rules.txt, whose header carries the whole
# design argument (why this is NOT a fourth rule, why the ids _on / _off were
# kept, and why no tnt_err_* key is born).
#
# THE NAMES ARE LOAD-BEARING, MECHANICALLY. _docs\tools\tnt_rule_conformance.ps1
# recognises a rule read as `<rule id>_trigger` (its Reads-Rule function), so
# calling these exactly tnt_trade_prestige_trigger / tnt_trade_piety_trigger is
# what keeps the section 6.13 check at exit 0 with no edit to its $RULES table.
# Rename either one and the tool goes red on twelve balancer rungs at once.
#
# WHO IS $A$ AND WHO IS $B$. $A$ is the side the question is asked FROM and $B$
# is the other party; on every live path that is { A = root  B = var:tnt_partner }
# because root is the player and var:tnt_partner is the counterparty in the
# window (E1 writes it first, tnt_30_effects.txt:90), on the letter path
# (tnt_ai_offer.0001 writes it in its immediate, before it composes) and hence
# inside the balancer, which both paths reach.
# BOTH HANDLES MUST BE ABSOLUTE - root, scope:x or var:... - NEVER `this` and
# NEVER `prev`. These are TEXT MACROS (the T51 rule, CLAUDE.md section 4): a
# relative handle re-binds inside the substituted body, and the body below
# changes scope with `$A$ = { ... }`.
#
# R13 AND THE UNSET PARTNER. Every relational rung opens with `exists = $A$` and
# `exists = $B$` before it touches a field, so a call made with no partner
# stored degrades to NO instead of throwing - the same answer the rule's `off`
# setting gives, which is the safe direction for every reader we have (a hidden
# row, a skipped balancer rung). The `_on` rung never touches $B$ at all.
# `exists = root` is real vanilla: GAME\common\scripted_triggers\00_coa_triggers.txt:129.
#
# R7 IS NOT ENGAGED. R7 forbids `var:a > var:b`, variable against variable.
# These rows compare a FIELD against the same FIELD reached through a scope, a
# shape vanilla writes five times over (citations at each row), and the mod
# already ships it: `root = { max_military_strength > var:tnt_partner.max_military_strength }`
# (common\scripted_guis\tnt_20_scripted_guis.txt:553) is the same dotted-var
# field read, in a shipped and exercised line.
#
# `$A$ = { <trigger> }` where $A$ expands to `root` is a legal scope change
# inside a trigger: GAME\common\scripted_effects\00_death_management_effects.txt:35
# (`limit = { is_alive = yes  root = { is_ai = yes } }`).
# =============================================================================
tnt_trade_prestige_trigger = {
	trigger_if = {
		# ALWAYS. The shipped default, so a player who never opens the lobby
		# sees exactly the mod he saw before W2.
		limit = { has_game_rule = tnt_trade_prestige_on }
		always = yes
	}
	trigger_else_if = {
		# PEERS AND BELOW ONLY. "I am at least as famed as he is."
		# REAL: `prestige_level >= scope:recipient.prestige_level`,
		#       GAME\common\character_interactions\00_fp3_interactions.txt:1478.
		# REAL in MACRO form: `prestige_level > $COMPARISON$.prestige_level`,
		#       GAME\common\scripted_triggers\00_laamp_triggers.txt:248
		#       (also 00_vassal_interactions.txt and 05_bp2_hostage_triggers.txt).
		limit = { has_game_rule = tnt_trade_prestige_peer }
		exists = $A$
		exists = $B$
		$A$ = { prestige_level >= $B$.prestige_level }
	}
	trigger_else = { always = no }		# tnt_trade_prestige_off, and any setting a later round adds without touching this trigger
}

tnt_trade_piety_trigger = {
	trigger_if = {
		limit = { has_game_rule = tnt_trade_piety_on }
		always = yes
	}
	trigger_else_if = {
		# SAME FAITH ONLY.
		# REAL: `faith = scope:actor.faith`,
		#       GAME\common\scripted_triggers\00_interaction_triggers.txt:296
		#       (grant_title_rivalry_trigger); also 00_councillor_triggers.txt.
		limit = { has_game_rule = tnt_trade_piety_faith }
		exists = $A$
		exists = $B$
		$A$ = { faith = $B$.faith }
	}
	trigger_else_if = {
		# SAME RELIGION - the same thought one step looser, so a Catholic and
		# an Orthodox still trade devotion and a Catholic and a Sunni do not.
		# REAL: `$WHO$.faith.religion = $TARGET$.faith.religion`,
		#       GAME\common\scripted_triggers\00_religious_triggers.txt:1181
		#       (is_heretic_trigger); negated at :1191 and :1204.
		limit = { has_game_rule = tnt_trade_piety_religion }
		exists = $A$
		exists = $B$
		$A$.faith.religion = $B$.faith.religion
	}
	trigger_else = { always = no }		# tnt_trade_piety_off
}

# Scope: LANDED TITLE. May $GIVER$ grant $RECEIVER$ a claim on this title?
#
# Deliberately symmetric with T3: the candidate pool is the GIVER'S OWN held
# titles, so granting a claim is a real concession (an invitation to press a
# claim against himself) rather than a free gift of someone else's land.
# Skipped if the receiver already holds a claim on it — that would be a no-op
# term with a non-zero price.
#
# `has_claim_on = prev` reaches back out to the title scope; NOT/AND/OR do not
# change scope. Proven verbatim:
# GAME\common\character_interactions\00_character_interactions.txt:860 and :1620
#   `has_claim_on = prev` inside a character block nested in a title scope
# (same shape at GAME\common\character_interactions\00_invite_agent_to_scheme.txt:221,
#  `scope:recipient = { has_claim_on = prev }`).
#
# v2 ITEM 1.1 GATE — "granting a claim on a title the partner already holds".
# With v1's independent-rulers-only partner set the two rows `holder ?= $GIVER$`
# and `NOT = { has_claim_on = prev }` were between them enough. They still are,
# strictly speaking — but the explicit `NOT = { holder ?= $RECEIVER$ }` row added
# below is what a reader of the send path can actually verify at a glance, and it
# is the row that stays correct if a future author ever widens the candidate pool
# beyond the giver's own titles (which is the obvious next request: "let me grant
# a claim on somebody ELSE's land"). Cost: one optional-link test per candidate.
#
# `?=` and not `=`: the receiver may hold nothing at all, and a title whose holder
# link is missing must not fail the whole trigger with a hard error.

# T5  tnt_can_vassalize_trigger = { LIEGE = <character>  VASSAL = <character> }
#
# Scope: character (root = the player). Can $VASSAL$ legally swear fealty to
# $LIEGE$?
#
# The tier requirement — a vassal must rank BELOW his liege — is the one place
# in this mod that wants a cross-character comparison. R0.3 forbids writing
# `$VASSAL$.highest_held_title_tier < $LIEGE$.highest_held_title_tier`, so it is
# folded through the named script value tnt_tier_edge_value (tnt_50_values.txt,
# contract §4.A) and compared against the literal 0:
#
#	tnt_tier_edge_value  =  root(player) tier  -  var:tnt_partner tier
#	> 0  ==>  the player outranks the partner
#	< 0  ==>  the partner outranks the player
#
# Because that value is hardwired to root-vs-partner, the trigger has to know
# WHICH WAY ROUND this call is. It reads that off $LIEGE$ rather than taking a
# third parameter, so the two callers keep the contract's parameter list:
#   tnt_vassal_p_toggle: LIEGE = var:tnt_partner, VASSAL = root  -> need < 0
#   tnt_vassal_r_toggle: LIEGE = root, VASSAL = var:tnt_partner  -> need > 0
#
# CROSS-FILE DEPENDENCY: root must be the player, and tnt_tier_edge_value must
# exist in tnt_50_values.txt. Both hold for every caller in this mod
# (scripted_gui is_valid runs with GuiScope.SetRoot( GetPlayer.MakeScope )).
#
# ---------------------------------------------------------------------------
# v2 ITEM 1.1 — THIS IS THE ITEM THAT NEEDED THE MOST TIGHTENING.
#
# Two of the nonsensical offers the widened partner set makes reachable are
# "vassalise your own vassal" and "swear fealty to your own vassal". Trace them:
#
#   tnt_vassal_r_toggle  (LIEGE = root, VASSAL = var:tnt_partner)
#     partner is already my vassal  ->  is_independent_ruler = no  ->  refused.
#   tnt_vassal_p_toggle  (LIEGE = var:tnt_partner, VASSAL = root)
#     I am already the partner's vassal -> I am not independent -> refused.
#     BUT: partner is MY vassal -> I am independent, so the only thing standing
#     in the way was the tier test, which a same-tier vassal passes (edge = 0
#     fails both `> 0` and `< 0`, so it happened to hold — by accident, not by
#     design). The explicit row
#         NOT = { $LIEGE$ = { is_vassal_or_below_of = $VASSAL$ } }
#     makes it a rule instead of a coincidence, in both directions.
#
# Everything else added below is lifted from the two vanilla interactions that do
# exactly these two things, so our gate cannot be looser than the engine's own:
#
#   offer_vassalization_interaction  GAME\common\character_interactions\00_character_interactions.txt:3-58
#     is_shown: recipient is_playable_character (:30), top_liege = this + is_tributary = no (:33-34),
#     highest_held_title_tier < actor's (:44), NOR cannot_be_vassal_or_liege / government_is_herder (:46-49)
#   offer_fealty_interaction         GAME\common\character_interactions\00_character_interactions.txt:1392-1436
#     is_shown: both sides is_landed + `top_liege = this` (:1401-1402, :1406-1407),
#     actor NOT cannot_be_vassal_or_liege (:1408), actor is_confederation_member = no (:1409)
#     is_valid: actor is_imprisoned = no, is_at_war = no (:1415-1416),
#     recipient NOR government_is_landless_adventurer / government_is_holy_order /
#     government_is_mercenary (:1430-1434)
#
# `is_independent_ruler = yes` is kept as our spelling of vanilla's
# `top_liege = this`; it is proven at
#   GAME\common\character_interactions\00_fp3_interactions.txt:1534,
#   GAME\common\character_interactions\00_grant_titles_interaction.txt:280, :1087,
#   GAME\common\character_interactions\00_tributary_interactions.txt:16.
#
# ONE DELIBERATE DEVIATION FROM VANILLA, flagged for the risk register.
# offer_fealty requires the would-be vassal to be at NO war at all
# (`is_at_war = no`, :1416). We narrow that to "not at war WITH THE FUTURE LIEGE"
# (`NOT = { is_at_war_with = ... }`), because ending a war by swearing fealty is
# the single most Total-War thing this mod can offer and vanilla's blanket ban
# would kill it. The narrower rule still forbids the one case that is genuinely
# ill-defined for `resolve_title_and_vassal_change`: a live war between the two
# parties whose vassal relation is being created. IF a live save ever shows a
# stuck war or an orphaned title change after a mid-war fealty deal, the revert
# is one line — replace the is_at_war_with row with vanilla's `is_at_war = no`
# inside the $VASSAL$ block.
# ---------------------------------------------------------------------------
# =============================================================================
tnt_can_vassalize_trigger = {
	trigger_if = {
		limit = {
			exists = $LIEGE$
			exists = $VASSAL$
		}
		NOT = { $LIEGE$ = { this = $VASSAL$ } }

		# v2 1.1: you cannot become the vassal of your own vassal, in either
		# direction. REAL: `is_vassal_or_below_of` — GAME\common\character_interactions\00_alliance.txt:2104, :2142
		NOT = { $LIEGE$ = { is_vassal_or_below_of = $VASSAL$ } }

		# v2 1.1: nor while the two of you are shooting at each other. See the
		# DELIBERATE DEVIATION note above. REAL: `is_at_war_with` —
		# GAME\common\character_interactions\00_alliance.txt:976, :1164
		NOT = { $VASSAL$ = { is_at_war_with = $LIEGE$ } }

		# AGOT SUSPENDED (CK3 1.20): compatibility guard parked; awaiting the AGOT update.
		# V14, USER ITEM 1: THE WALL IS NOT A PARTY TO ANYONE'S FEALTY, IN
		# EITHER DIRECTION. Full argument and the vanilla-safety proof are in
		# T46's header below; AGOT gates the two vanilla interactions this
		# trigger was lifted from with the same test in the same both-sides
		# shape - AGOT\common\character_interactions\00_character_interactions.txt
		# :63-66 (offer_vassalization) and :1547-1550 (offer_fealty).
		# ONE ROW, SIX DOORS: the two show gates (tnt_41_gates.txt:1689, :1703),
		# the apply leg (tnt_32_apply.txt:1535), A14 (tnt_37_ai_offer.txt:2003),
		# A20 (:3004) and the auto-balance filler (tnt_39_autobalance.txt:2351)
		# all delegate here. In a vanilla game the row is always true.
		# T15 below carries the matching row - its own header makes row-for-row
		# parity with this trigger a stated invariant.
		# tnt_vassalage_pair_allowed_trigger = { A = $LIEGE$  B = $VASSAL$ }

		$VASSAL$ = {
			is_alive = yes
			is_landed = yes
			is_independent_ruler = yes		# only an independent ruler has fealty of his own to give. REAL: GAME\common\character_interactions\00_fp3_interactions.txt:1534, 00_grant_titles_interaction.txt:280
			is_imprisoned = no			# v2 1.1. REAL: GAME\common\character_interactions\00_character_interactions.txt:1415 (offer_fealty), 00_alliance.txt:1158
			is_confederation_member = no		# v2 1.1. REAL: GAME\common\character_interactions\00_character_interactions.txt:1409 (offer_fealty is_shown); the `= yes` form at :63
			NOT = { government_has_flag = cannot_be_vassal_or_liege }	# v2 1.1. REAL: GAME\common\character_interactions\00_character_interactions.txt:1408
		}
		$LIEGE$ = {
			is_alive = yes
			is_landed = yes
			# V11, USER ITEM 6: THE VASSAL-LIMIT ROW IS GONE. It was ours, not
			# vanilla's — neither offer_fealty_interaction
			# (00_character_interactions.txt:1392-1436, read in full: is_shown has
			# the two top_liege rows, the tier row and the actor's government rows;
			# is_valid_showing_failures_only has is_imprisoned, is_at_war, the
			# proximity description and the recipient's government NOR) nor
			# offer_vassalization_interaction (:3-58) carries one, and the header
			# above enumerates every vanilla row this trigger lifted — the limit was
			# never among them. Being over the vassal limit is a designed, survivable
			# CK3 state that costs opinion, not a legality bar, and as a hard row it
			# silently deleted the whole fealty archetype for any player at his cap,
			# with no message anywhere, because A14 never even reached the letter
			# (0 arch=14 compositions in 47 draws, 2026-08-16). tnt_exec_vassal_effect
			# (tnt_32_apply.txt:1341-1350) re-validates this same trigger and needs
			# nothing from it: resolve_title_and_vassal_change does not refuse an
			# over-limit liege. T15's reporting twin lost the matching
			# tnt_err_vassal_limit row in the same commit, so the window's toggle and
			# this gate still ask exactly the same question.
			# v2 1.1: a mercenary captain, a holy order and a landless adventurer
			# cannot hold anyone's fealty. REAL, as a NOR on the prospective liege:
			# GAME\common\character_interactions\00_character_interactions.txt:1430-1434
			NOR = {
				government_has_flag = cannot_be_vassal_or_liege
				government_has_flag = government_is_landless_adventurer
				government_has_flag = government_is_holy_order
				government_has_flag = government_is_mercenary
			}
		}
		# --- the tier test, direction-aware, see the header comment above ---
		trigger_if = {
			limit = { $LIEGE$ = { this = root } }
			tnt_tier_edge_value > 0			# player is liege: the partner must rank below him
		}
		trigger_else = {
			tnt_tier_edge_value < 0			# partner is liege: the player must rank below him
		}
	}
	trigger_else = { always = no }
}

# banners) are already vanilla's job above, so the extra line only hid the most
# interesting objects in the game from the negotiating table.

# is the correct answer for a legendary sword, not a reason to hide it.
# Rarity keys verified at GAME\common\script_values\00_artifact_values.txt:13,22,31.

tnt_artifact_giftable_trigger = {
	trigger_if = {
		limit = { exists = $RECEIVER$ }
		artifact_can_be_gift_to_trigger = { RECIPIENT = $RECEIVER$ }
	}
	trigger_else = { always = no }

# Rule R14 says everything user-visible needs a loc key, and the point of the

# T49b  tnt_ai_coercive_fealty_pair_trigger = { A = <hegemon> B = <victim> }
#
# Strategic threat is deliberately a separate road from both the cash ultimatum
# and consensual S-FEALTY. It is reachable only for an independent king/emperor
# with a real field army and free vassal capacity, against a peaceful, playable,
# lower-tier independent AI neighbour. Both rulers must be at peace: changing a
# liege in the middle of an ordinary war is unsafe, and in AGOT Mega Wars the
# temporary `agot_pwl_direct` independence is explicitly not a real vassalization
# opportunity. The recent-independence opinion lock mirrors vanilla's interaction
# validity and prevents an immediate undo of a granted release. The shared threat
# gate adds independence, no alliance or hostage link, and the fifteen-year pair
# memory. Submission has BOTH a raw 4:1 max-strength floor and 100 final threat
# points: prestige/dread may strengthen the instrument, but may never turn a merely
# prestigious near-peer into an "overwhelming" hegemon. The 2026-09-30 access
# ruling applies to every threat. At normal N=5, equal prestige/dread needs
# a 21:1 ratio for 100 points; the separate raw 4:1 floor still applies.
#
# $A$ and $B$ MUST BOTH BE ABSOLUTE HANDLES on both live call sites. T49 reads
# root as proposer, so the caller names the iterator candidate before calling;
# passing relative `this` here would rebind inside the $A$ scope and waste a pulse.
# =============================================================================
tnt_ai_coercive_fealty_pair_trigger = {
	trigger_if = {
		limit = {
			exists = $A$
			exists = $B$
		}
		$A$ = {
			is_ai = yes
			is_alive = yes
			is_landed = yes
			is_ruler = yes
			is_adult = yes
			is_independent_ruler = yes
			is_at_war = no
			highest_held_title_tier >= tier_kingdom
			vassal_limit_available > 0
			NOT = { has_variable = tnt_ai_world_cd }
			max_military_strength > 0
			current_military_strength >= fifty_percent_of_max_military_strength
			NOT = { has_truce = $B$ }
			# AGOT SUSPENDED (CK3 1.20): Mega-War exclusion parked; awaiting the AGOT update.
			# NOT = { has_variable = agot_pwl_direct }
		}
		$B$ = {
			tnt_ai_world_partner_structural_trigger = yes
			is_independent_ruler = yes
			is_at_war = no
			is_playable_character = yes
			is_tributary = no
			is_confederation_member = no
			# AGOT SUSPENDED (CK3 1.20): Mega-War exclusion parked; awaiting the AGOT update.
			# NOT = { has_variable = agot_pwl_direct }
			NOT = {
				has_opinion_modifier = {
					modifier = granted_independence_opinion
					target = $A$
				}
			}
			highest_held_title_tier < $A$.highest_held_title_tier
			NOR = {
				government_has_flag = cannot_be_vassal_or_liege
				government_has_flag = government_is_herder
			}
		}
		# AGOT SUSPENDED (CK3 1.20): compatibility guard parked; awaiting the AGOT update.
		# tnt_vassalage_pair_allowed_trigger = { A = $A$  B = $B$ }
		tnt_can_threat_trigger = { A = $A$  B = $B$ }
		$A$ = { tnt_threat_ratio_value >= 4 }
		$A$ = { tnt_threat_points_value >= 100 }
	}
	trigger_else = { always = no }
}

# T50  tnt_ai_world_title_saleable_trigger    NO PARAMETERS     <- NEW in V16
#
# Scope: A LANDED TITLE. scope:actor MUST be the buyer and scope:recipient the
# seller — i.e. it is callable only from inside the AI-to-AI pulse's chain
# (common\scripted_effects\tnt_38_ai_world.txt), which is where both scopes live.
#
# May the actor buy THIS title from the recipient in a silent AI-to-AI treaty?
# It is the whole of S-TITLE's eligibility: county tier, T3, and the reach gate.
#
# WHY IT EXISTS (V16, user item 2, finding Q2-F5). E20's entry trigger asked only
# for a cedable county; its PICK asked for a cedable county ON THE BUYER'S REACH.
# So the archetype could win the draw and then find nothing, and it did: of 227
# arch=title draws in the 2026-08-17 run, 134 ended WSKIP why=no_eligible_title
# (59%), which E20's own comment already diagnosed as "the entry trigger proved a
# cedable county EXISTS, so this means the reach gate filtered them all". Each of
# those 134 draws also spent 22 of the sandbox draw's 100 weight points on
# nothing. THE INVARIANT THE LETTER PATH STATES FOR ITSELF IS NOW TRUE HERE TOO
# (tnt_37_ai_offer.txt, A10's pass 3: "the LAST pass's limit must be the entry
# trigger's own body"). One definition, called from both places, so they cannot
# drift.
#
# THIS IS LEGALITY, NOT A BALANCE, so it is inside E12's rule for entry triggers
# ("CHEAP LEGALITY TESTS ONLY, never a balance", tnt_38_ai_world.txt:1109-1114).
# THE COST IS STATED HONESTLY: the reach gate now runs on every firing pulse
# instead of on the 22% that draw S-TITLE. It is a bounded MAP walk and not a
# value chain, which is what that rule is about, and it is cheap in the common
# case because the de jure disjunct comes FIRST and is one flat comparison —
# `any_neighboring_county` is only reached when the buyer is not the de jure
# liege, `any_held_title` short-circuits on the first county that passes, and the
# two parties are neighbouring realms by construction of the partner pick. In
# exchange, the identical walk stops running inside a doomed pick 134 times a
# half-year.
#
# THE `holder` READ IS GUARDED, WHICH THE INLINE VERSION WAS NOT. E20's pick used
# `OR = { holder = scope:actor  holder = { any_liege_or_above = { this =
# scope:actor } } }` — an UNGUARDED dereference of a neighbouring county's holder,
# evaluated once per (held county x neighbouring county). The letter path already
# replaced all six copies of that same body for this reason and records why
# (tnt_37_ai_offer.txt, the R13 paragraph above A8's adjacency row): a
# neighbouring county with no holder is not hypothetical, and AGOT fills its
# unowned map with placeholder holders. Vanilla's own trigger for this exact
# iterator body opens `holder ?= { … }` and is semantically identical
# (`this = $TARGET$` OR `target_is_liege_or_above = $TARGET$` versus
# `any_liege_or_above = { this = … }`):
#   GAME\common\scripted_triggers\00_title_triggers.txt:153-159
#     county_held_or_vassal_to_target_trigger = { TARGET = … }
# It is a VANILLA scripted trigger, so it is mod-load safe with and without AGOT.
#
# ROW-BY-ROW EVIDENCE (all four rows were already shipped and cited at E20; this
# trigger moves them, it does not invent them):
#   tier = tier_county                 one county is a border adjustment, a duchy
#                                      is a war — E20's own gate 1
#   tnt_can_cede_title_trigger         T3, which itself refuses the seller's
#                                      primary title and capital county
#   target_is_de_jure_liege_or_above   REAL on a title scope with a title target,
#     GAME\common\casus_belli_types\09_mpo_wars.txt:3672
#   any_neighboring_county             REAL, and paired with its across-water
#     sibling in vanilla's own border test,
#     GAME\common\schemes\scheme_types\ep3_dispute_border_scheme.txt:36-46
#
# DELIBERATELY NOT ADDED: the across-water door. E20's gate is the LAND border on
# purpose (a silent land transfer over a strait is a different design question
# from a letter the player can refuse), and widening it here would change the
# archetype's meaning inside a commit whose job is to make the two ends agree.
# The letter path's A8/A10 keep their paired-OR; this one keeps its single door.
# =============================================================================
tnt_ai_world_title_saleable_trigger = {
	tier = tier_county
	tnt_can_cede_title_trigger = { GIVER = scope:recipient }
	# The reach gate, both honest readings (item 7b, E20 header gate 3):
	# de jure tidy-up OR a county on the buyer's own frontier.
	OR = {
		target_is_de_jure_liege_or_above = scope:actor.primary_title
		any_neighboring_county = {
			county_held_or_vassal_to_target_trigger = { TARGET = scope:actor }
		}
	}
}
