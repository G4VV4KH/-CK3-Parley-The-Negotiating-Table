# Bound in the GUI to button_round wrappers that MUST carry `button_ignore = none`,

# ONE CLAMP PER ENTRY SINCE USER ITEM 4, not two. Item 14 added the AFFORDABILITY
# cap next to the old 5000 SANITY literal; item 4 then REMOVED every gold/prestige/
# piety sanity literal mod-wide — the giver's actual stock is the only ceiling a
# deal amount has now. (Influence alone keeps its 1000 literal, tnt_22_v2.txt —
# the user's ruling named gold 5000 and prestige/piety 10000, not influence.)
# The affordability cap: `max = gold` off root on the _p side,
# `max = scope:tnt_cap_giver.gold` off var:tnt_partner on the _r side, each behind
# a `gold >= 1` ladder. THE LADDER IS THE POINT: gold goes negative in CK3, so a
# bare `max = gold` would clamp the row to a NEGATIVE maximum. Below 1 gold the
# amount is forced to 0. See the long note inside tnt_gold_p_add and the one above
# tnt_gold_r_add.
#
# Shape note (deliberate hardening, no frozen name changed):
# tnt_reset_offer_effect removes the 24 offer variables, so a stepper is routinely the FIRST
# thing to touch its variable. Vanilla always guards change_variable with `exists = var:x`
# (GAME\common\focuses\00_education_focuses.txt:49-56,
#  GAME\common\scripted_effects\10_dlc_tgp_scripted_effects.txt:1596-1605), so the create case
# goes through set_variable instead. The clamp line afterwards is the contract's verbatim form;
# the value/add/subtract/multiply/divide/min/max/round vocabulary is confirmed at
# GAME\common\scripted_effects\00_diarchy_scripted_effects.txt:1002-1011.

tnt_gold_p_add = {
	scope = character

	effect = {
		if = {
			limit = { exists = var:tnt_gold_p }
			change_variable = {
				name = tnt_gold_p
				add = 10
			}
		}
		else = {
			set_variable = {
				name = tnt_gold_p
				value = 10
			}
		}
		# AFFORDABILITY (user item 14). Root is the player, and the player is the giver
		# on the _p side, so the stock is read bare off root. `max = gold` bare on root
		# is vanilla-proven: GAME\common\script_values\03_dlc_fp2_script_values.txt:235
		# and GAME\common\script_values\07_ep3_values.txt:4938.
		#
		# THE `>= 1` LADDER IS NOT OPTIONAL AND MUST NEVER BE FLATTENED TO A BARE
		# `max = gold`. Gold goes NEGATIVE in CK3, and a negative max would clamp the
		# row to a negative maximum. A giver below 1 gold lands on the else branch and
		# the amount is forced to 0 — the same convention the 24 prestige/piety/
		# influence steppers use (tnt_22_v2.txt, groups V1a-V1c). `min` nested inside a
		# `max = { }` block has ZERO vanilla call sites, which is why it is a ladder.
		#
		# tnt_send_offer.is_valid stays the second line of defence (tnt_gold_shortfall_value
		# below): the stock can fall between composing the offer and pressing Propose.
		if = {
			limit = { gold >= 1 }
			set_variable = {
				name = tnt_gold_p
				value = {
					value = var:tnt_gold_p
					min = 0
					max = gold
				}
			}
		}
		else = {
			set_variable = {
				name = tnt_gold_p
				value = 0
			}
		}
	}
}
