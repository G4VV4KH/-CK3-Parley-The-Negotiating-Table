# Legal keys only: scope, saved_scopes, effect, the shown-gate and is_valid. The shown-gate is
# used by no entry in this mod. is_valid appears on exactly 2 entries here
# (tnt_pick_title_p_open, tnt_pick_title_r_open — ceding a title is game-rule gated).
#
# HOW A PICKER WORKS, end to end
#   1. "Choose..." button -> an OPENER (group C). It writes tnt_picker = <item index, R0.8>
#      and tnt_picker_side (0 = _p, 1 = _r), then calls the matching fill effect so the
#      variable list is current. MakeScope.GetList is a saved variable list, not a live query.
#   2. The overlay shows because tnt_picker > 0; the individual panel shows because
#      EqualTo_CFixedPoint( TntNum('tnt_picker'), '<index>' ) and the same test on
#      tnt_picker_side. No GetScriptedGui(...) is ever called from a visible.
#   3. A row click -> a SETTER (group D). The row object arrives as scope:tnt_picked through
#      exactly ONE .AddScope('tnt_picked', <X>.MakeScope) — two chained AddScope calls are
#      unverified anywhere and must never be written.
#      The setter stores the object and removes tnt_picker, which closes the panel.
#   4. The row's selected state is ours alone:
#      down = "[ObjectsEqual( TntVar('tnt_title_p').Title, Title.Self )]".
#      Title.IsSelected / Character.IsSelected mean MAP selection and are not usable here.
#   5. The little X on a filled row -> a CLEARER (group E). One line each, deliberately
#      dumber than a toggle so that no object-equality test is ever needed.
#
# Item indices (R0.8): 6 = title, 10 = artifact.
# 7 (claim, V13 user item 1), 8 (prisoners) and 11 (nonwar) are RETIRED, not free:
# never renumbered, never reused, exactly as index 2 (the yearly tribute) has been
# since it was withdrawn. The full retired-not-free list after V13 is
# 2, 3, 4, 7, 8, 11, 13.
#
#-------------------------------------------------------------------------------------

# Cede a title, player's side: candidates are the PLAYER's held titles, tier >= county,
# excluding the primary title and the capital (tnt_can_cede_title_trigger does the filtering).
tnt_pick_title_p_open = {
	scope = character

	is_valid = {
		custom_description = {
			text = tnt_err_advanced_off
			exists = var:tnt_advanced
		}
	}

	effect = {
		set_variable = {
			name = tnt_picker
			value = 6
		}
		set_variable = {
			name = tnt_picker_side
			value = 0
		}
		tnt_fill_title_p_effect = yes
		# ITEM 3 — THE ORDER. These four v1 single-object openers share their
		# candidate lists with the multi-select panel, so a fill here would leave
		# tnt_list_title_p in engine order while every multi-select entry keeps it in
		# key order — two panels, two orders, one list. The rule this file now obeys
		# with the rest of the subsystem is simply "every fill is followed by a
		# sort"; argument in common\scripted_effects\tnt_35a_sort.txt.
		tnt_sort_all_effect = yes
	}
}

# GROUP D — PICKER SETTERS (6, was 8 before the two nonwar setters retired)
#
# saved_scopes = { tnt_picked } on all 6, and on these 6 only.
# GUI macro used per type (exactly one AddScope in each):
#   titles    -> TntPickTitle('...')     AddScope( 'tnt_picked', Title.MakeScope )
#   artifacts -> TntPickArtifact('...')  AddScope( 'tnt_picked', Artifact.MakeScope )
#   rulers    -> TntPickChar('...')      AddScope( 'tnt_picked', Character.MakeScope )
#
# The `exists = scope:tnt_picked` guard costs nothing and keeps error.log clean if the row's
# object is gone by the time the click lands (report 02 §2.3 recommends guarding GUI-supplied
# scopes for exactly this reason).
#
# Removing tnt_picker closes the panel, because the panel's visible tests tnt_picker.
# tnt_picker_side is left alone here — it is harmless residue, is overwritten by the next
# opener, and is removed by tnt_reset_offer_effect and by tnt_picker_close.
######################################################################################

tnt_pick_title_p_set = {
	scope = character
	saved_scopes = { tnt_picked }

	effect = {
		if = {
			limit = { exists = scope:tnt_picked }
			set_variable = {
				name = tnt_title_p
				value = scope:tnt_picked
			}
		}
		remove_variable = tnt_picker
	}
}

tnt_pick_title_r_set = {

}
