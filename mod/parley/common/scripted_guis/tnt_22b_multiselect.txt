# PARLEY: THE NEGOTIATING TABLE — MULTI-SELECT SCRIPTED GUIS  (user item 2.2)
#
# Owner: multiselect.  24 entries, in four groups of six:
#   A  openers   tnt_multi_<item>_<side>_open      (6)
#   B  add       tnt_sel_<item>_<side>_add         (6)
#   C  remove    tnt_sel_<item>_<side>_remove      (6)
#   D  clear     tnt_sel_<item>_<side>_clear       (6)
# where <item> is title | claim | artifact and <side> is p | r.
#
# Legal keys only: scope, saved_scopes, is_shown, is_valid, effect
# (00-ARCHITECTURE §3.6 settles the set; ai_is_valid, confirm_title,
# ai_chance, is_hidden, alert and on_click do NOT exist). is_shown is used
# by no entry here. is_valid is used by exactly two — the two title
# openers, because ceding a title is game-rule gated, which is exactly
# what the shipped tnt_pick_title_p_open / _r_open do
# (tnt_21_pickers.txt:85-90, :109-114).
#
#-------------------------------------------------------------------------------------
# HOW A MULTI-SELECT PANEL WORKS, END TO END
#-------------------------------------------------------------------------------------
#  1. The item row's "Choose..." button calls an OPENER (group A). It writes
#     tnt_picker = <frozen item index, contract R0.8> and tnt_picker_side
#     (0 = _p, 1 = _r), rebuilds the available list, and then PRUNES the
#     objects that are already chosen out of it. Order matters: the fill
#     effects in tnt_31_lists.txt open with clear_variable_list and rebuild
#     from the world, so without the prune every already-chosen title would
#     reappear in the "available" column and be shown twice.
#  2. gui/tnt_panel_multiselect.gui's panel shows because tnt_picker and
#     tnt_picker_side match its two EqualTo_CFixedPoint tests. No
#     GetScriptedGui is called from any visible, anywhere (Law R11).
#  3. A click in the AVAILABLE column calls an ADD (group B). A click in the
#     CHOSEN column calls a REMOVE (group C). Both receive the row object as
#     scope:tnt_picked through exactly ONE .AddScope — chaining two is
#     unverified and forbidden (Law R8). Both leave tnt_picker SET, so the
#     panel stays open for the next pick. That is the one behavioural
#     difference from the shipped single-object setters, which end with
#     `remove_variable = tnt_picker` because one pick is the whole
#     interaction.
#  4. The panel's Done button and its close button both call the SHIPPED
#     tnt_picker_close (tnt_21_pickers.txt), which clears tnt_picker and
#     tnt_picker_side. This file adds no closer of its own.
#  5. There is NO `down =` on any row and NO membership test anywhere,
#     because a row's COLUMN is its state. The engine has no list-membership
#     datafunction; that absence is what the two-list shape is for.
#
#-------------------------------------------------------------------------------------
# WHAT THIS FILE SUPERSEDES (nothing is deleted anywhere)
#-------------------------------------------------------------------------------------
# tnt_21_pickers.txt:33-71 predicted this file entry by entry. Its own audit
# says: the 10 openers "ALL SURVIVE UNCHANGED", 6 of its 8 setters are
# "SUPERSEDED, not yet dead", and 6 of its 9 clearers likewise. This file
# does NOT touch that file (Law R18, exclusive ownership). Instead it adds
# parallel entries under new names, and the GUI owner re-points six buttons:
#     tnt_pick_title_p_open     ->  tnt_multi_title_p_open
#     tnt_pick_title_r_open     ->  tnt_multi_title_r_open
#     tnt_pick_claim_p_open     ->  tnt_multi_claim_p_open   BOTH NAMES NOW RETIRED
#     tnt_pick_claim_r_open     ->  tnt_multi_claim_r_open   BOTH NAMES NOW RETIRED
#       (V13, user item 1: the granted claim is a design refusal. The supersession
#        happened and then the whole family was withdrawn, so NEITHER generation is
#        live. See TOMBSTONE C in common\scripted_effects\tnt_32_apply.txt.)
#     tnt_pick_artifact_p_open  ->  tnt_multi_artifact_p_open
#     tnt_pick_artifact_r_open  ->  tnt_multi_artifact_r_open
# Until that swap lands, the shipped single-object path keeps working and
# nothing here is reachable — so this file can ship before the GUI does.
# The six superseded setters and six superseded clearers stay live and
# harmless; they write tnt_title_p and friends, which nothing reads once the
# valuation and the deal sheet point at the selection lists. A later cleanup
# pass may delete them, together with the tnt_title_* rows in
# tnt_reset_offer_effect and the dead re-validation rows in
# tnt_send_offer's is_valid.
#
#-------------------------------------------------------------------------------------
# COST
#-------------------------------------------------------------------------------------
# Nothing in this file is on a hot path. The only is_valid rows are the two
# title openers' single `exists = var:tnt_advanced` test, evaluated once per
# frame on ONE button each — the same test the shipped title openers already
# carry. No entry here is bound to a per-row visible or down (Law R11), and
# the panel's per-frame cost is four script values (two counts, one cap flag,
# and the domain pair inside the title panels), read once per panel, never
# once per row.
######################################################################################

# Titles the PLAYER cedes. tnt_can_cede_title_trigger inside
# tnt_fill_title_p_effect does the filtering (own held titles, county or
# above, not the primary title, not the capital).
#
# Game-rule gated, exactly like the shipped opener it replaces: ceding land
# is one of the three Tier-4 terms behind tnt_advanced (the others are
# marriage and fealty). tnt_advanced is written once per window opening by
# tnt_open_window_effect, so this is a map lookup and not a trigger tree.
tnt_multi_title_p_open = {
	scope = character

	is_valid = {
		custom_tooltip = {
			text = tnt_err_advanced_off
			exists = var:tnt_advanced
		}
	}

	effect = {
		set_variable = {
			name = tnt_picker
			value = 6
		}
		set_variable = {
			name = tnt_picker_side
			value = 0
		}
		tnt_fill_title_p_effect = yes
		tnt_multi_prune_effect = { SEL = tnt_sel_title_p  LIST = tnt_list_title_p }
		# ITEM 3 — THE ORDER. The fill above rebuilds the candidate list from the
		# world in engine order, so the sort has to follow it, and it has to follow
		# the prune too (the prune only shortens). One line, no parameters, and
		# idempotent — every mutation site in the subsystem carries this same line;
		# the whole argument is in common\scripted_effects\tnt_35a_sort.txt.
		tnt_sort_all_effect = yes
	}
}

# GROUP B — THE SIX "ADD" SETTERS   (a click in the AVAILABLE column)
#
# saved_scopes = { tnt_picked } on all six. GUI macro per type, one AddScope
# each (tnt_macros.txt:78-94):
#   titles / claims -> TntPickTitle('tnt_sel_title_p_add')
#   artifacts       -> TntPickArtifact('tnt_sel_artifact_p_add')
#
# The whole body is one call into tnt_multi_add_effect, which owns the
# existence guard and the duplicate guard. IT NO LONGER OWNS A CAP: the cap of
# 3 was deleted from it in V4 (tnt_35_multiselect.txt:124-137), so claims and
# artifacts are as unbounded as titles — and the four claim/artifact cap FLAGS
# have followed it to hard zeros (tnt_54_multiselect_values.txt §2: they kept
# firing at >= 3 and lighting tnt_err_too_many, whose domain-limit wording was
# the reported "artifacts count toward the domain limit" bug). Keeping the
# logic in ONE effect rather than six copies is why deleting the cap was a
# four-line change. *** v3 ITEM 5: THE TWO TITLE SETTERS DO NOT USE IT — see
# the box above them. Claims and artifacts still do. (The captive list did too,
# until the prisoner terms were RETIRED — tombstone at E22 in
# common\scripted_effects\tnt_32_apply.txt. The shared effects themselves are
# untouched: they are also the courtier transfer's machinery.) ***
#
# NO is_valid. A click is never refused, and that is deliberate twice over:
# there is no cap left to refuse it, and an is_valid here would have to be
# evaluated by the GUI per row per frame to grey the rows, which is exactly
# what Law R11 forbids.
######################################################################################

######################################################################################
# *** THE TWO TITLE ADD SETTERS DO NOT GO THROUGH tnt_multi_add_effect ANY MORE ***
# v3 ITEM 5. The user: "let us remove the limit, and instead make exceeding the
# domain limit give fewer points."
#
# tnt_multi_add_effect's SECOND guard WAS the cap —
#     trigger_if = { limit = { has_variable_list = $SEL$ }
#                    NOT = { variable_list_size = { name = $SEL$  value >= 3 } } }
# — and there was no parameter to switch it off for titles alone, which is why
# these two setters stopped calling the shared body and wrote it out by hand.
# V4 then deleted that guard from the shared body too
# (tnt_35_multiselect.txt:124-137), so claims and artifacts are now just as
# uncapped, and the difference between these two hand-written setters and the
# shared effect is history, not behaviour.
#
# The remaining two guards ARE reproduced below, verbatim in shape and in order,
# because both are load-bearing:
#   1. `exists = scope:tnt_picked` — the title may have been destroyed between the
#      row being laid out and the click landing;
#   3. the duplicate guard, wrapped in `trigger_if = { limit = { has_variable_list …`
#      because on the FIRST pick of a session the list does not exist at all and
#      is_target_in_variable_list would then be asked about an unset list (defect
#      D5's class of error). trigger_if simply does not fire and the pick succeeds,
#      which is the correct answer. Same shape as tnt_31_lists.txt:301-313.
# and so is the second half of the MOVE — removing the object from the AVAILABLE
# list, guarded so that nothing is asked of a list that does not exist.
#
# tnt_picker is deliberately LEFT SET, exactly as tnt_multi_add_effect leaves it:
# a multi-select panel must stay open so the player can pick the next title.
#
# WHAT NOW BOUNDS THE TITLE LISTS: NOTHING AT ALL, and that is deliberate as of
# user item 13. The price that used to bound them (tnt_domain_penalty_p/_r_value,
# common\script_values\tnt_54_multiselect_values.txt §4b) has been deleted, so
# going over the domain limit no longer costs a single point. The only thing left
# is the WARNING: §4 of that file still computes tnt_domain_over_p/_r_value and the
# panels still colour their two counter lines red. A player may put his whole realm
# on the table and will be told so, but not charged for it.
######################################################################################

tnt_sel_title_p_add = {
	scope = character
	saved_scopes = { tnt_picked }

	effect = {
		if = {
			limit = {
				exists = scope:tnt_picked

				trigger_if = {
					limit = { has_variable_list = tnt_sel_title_p }
					NOT = {
						is_target_in_variable_list = {
							name = tnt_sel_title_p
							target = scope:tnt_picked
						}
					}
				}
			}

			add_to_variable_list = {
				name = tnt_sel_title_p
				target = scope:tnt_picked
			}

			if = {
				limit = {
					has_variable_list = tnt_list_title_p
					is_target_in_variable_list = {
						name = tnt_list_title_p
						target = scope:tnt_picked
					}
				}
				remove_list_variable = {
					name = tnt_list_title_p
					target = scope:tnt_picked
				}
			}
		}

		# ITEM 3 — THE ORDER, and it is OUTSIDE the if on purpose: a refused click
		# (a double click, or a row the prune has not caught) then still leaves both
		# columns in the one canonical order instead of in click order. Idempotent,
		# so being called on a no-op costs one guard that fails.
		tnt_sort_all_effect = yes
	}
}

# GROUP D — THE SIX "CLEAR THIS TERM" ENTRIES
#
# Bound to the little X in the item row (tooltip tnt_btn_none), the same
# affordance the shipped single-object clearers use, and to the panel's own
# "clear" control if the GUI owner wants one.
#
# Two lines each, and the second one is the interesting one: instead of
# handing every chosen object back individually, the available list is
# simply REBUILT from the world by the shipped fill effect. With the
# selection list now empty there is nothing to prune, so fill alone restores
# the exact correct "available" set — including dropping anything that has
# become uncedeable in the meantime. That is strictly more correct than a
# hand-back loop and it is two lines instead of ten.
#
# The names deliberately do NOT collide with the shipped clearers
# (tnt_title_p_clear, tnt_claim_p_clear, tnt_artifact_p_clear) — those
# remove the single-object variables and stay live until the cleanup pass.
######################################################################################

	scope = character

	effect = {
		clear_variable_list = tnt_sel_title_p
		tnt_fill_title_p_effect = yes
		# ITEM 3. The clear leaves nothing to order, but the FILL beside it rebuilds
		# the candidate column from the world in engine order, so this line is not
		# optional here either.
		tnt_sort_all_effect = yes
	}
}
