###############################################################################
# PART 1 — THE FIVE BASE VALUES
#
# The ONLY five values in this file that touch a scope. Everything below chains
# these by name. Each is guarded with `if = { limit = { exists = scope:… } … }`, the
# BP2 template's own shape, so a stray evaluation outside an interaction returns the
# floor instead of an error line.
###############################################################################

# --- yearly income, floored at 12 so a pauper's ratio never explodes ----------
# yearly_character_income as a script-value field: REAL,
# GAME\common\script_values\01_dynamic_values.txt:316
#   gift_interaction_cutoff = { value = yearly_character_income  multiply = 3  min = 10  max = 300 }
# Floor of 12 copied from tnt_actor_year_income_value (tnt_50_values.txt:281-285).
tnt_ai_actor_income_value = {
	value = 0
	if = {
		limit = { exists = scope:actor }
		scope:actor = { add = yearly_character_income }
	}
	min = 12
}

tnt_ai_recipient_income_value = {
	value = 0
	if = {
		limit = { exists = scope:recipient }
		scope:recipient = { add = yearly_character_income }
	}
	min = 12
}

# --- military strength, floored at 1 -----------------------------------------
# current_military_strength as a script-value field: REAL,
# GAME\common\script_values\00_interaction_values.txt:824;
# also GAME\common\character_interactions\00_alliance.txt:2009 inside ai_accept.
tnt_ai_actor_strength_value = {
	value = 0
	if = {
		limit = { exists = scope:actor }
		scope:actor = { add = current_military_strength }
	}
	min = 1
}

tnt_ai_recipient_strength_value = {
	value = 0
	if = {
		limit = { exists = scope:recipient }
		scope:recipient = { add = current_military_strength }
	}
	min = 1
}

# --- signed tier edge: > 0 means the ACTOR outranks the recipient -------------
# Exists so that nothing anywhere compares two characters' fields directly
# (contract R0.3 / Law 4, and R7 of the build brief). Sign convention deliberately
# matches tnt_tier_edge_value (tnt_50_values.txt:400-404) with root -> scope:actor
# and var:tnt_partner -> scope:recipient, so "< 0 means the decider is the higher
# tier" reads the same in both families.
# highest_held_title_tier as a script-value field: REAL,
# GAME\common\script_values\00_title_tiers_values.txt:10.
tnt_ai_tier_edge_value = {
	value = 0
	if = {
		limit = { exists = scope:actor }
		scope:actor = { add = highest_held_title_tier }
	}
	if = {
		limit = { exists = scope:recipient }
		scope:recipient = { subtract = highest_held_title_tier }
	}
}


# Telemetry-only twin for the strategic submission lane. It intentionally reads
# the lane's private target scope, so a failed coercion search can never seed the
# generic sandbox's scope:recipient.
tnt_ai_submission_tier_edge_value = {
	value = 0
	if = {
		limit = { exists = scope:actor }
		scope:actor = { add = highest_held_title_tier }
	}
	if = {
		limit = { exists = scope:tnt_submission_target }
		scope:tnt_submission_target = { subtract = highest_held_title_tier }
	}
}


###############################################################################
# PART 2 — THE TWO POWER RATIOS
#
# Shape copied from vanilla's own military-balance modifier,
# GAME\common\character_interactions\00_alliance.txt:2005-2014, and from the window's
# tnt_power_ratio_value / tnt_power_ratio_inv_value (tnt_50_values.txt:372-385).
# Clamped 0.2..5 so one enormous or one empty realm cannot dominate a price.
###############################################################################

# How strong the ACTOR is relative to the recipient. Used where the recipient is
# valuing something the ACTOR brings (an alliance with him, a truce with him).
tnt_ai_power_ratio_value = {
	value = tnt_ai_actor_strength_value
	divide = { value = tnt_ai_recipient_strength_value  min = 1 }
	min = 0.2
	max = 5
}

# The mirror. Used where the recipient is valuing something HE gives up (his own
# freedom to attack, his own obligation to march).
tnt_ai_power_ratio_inv_value = {
	value = tnt_ai_recipient_strength_value
	divide = { value = tnt_ai_actor_strength_value  min = 1 }
	min = 0.2
	max = 5
}


###############################################################################
# PART 3 — THE AMOUNTS, IN RAW RESOURCE
#
# These are what the effects actually hand over: gold and prestige.
# Every one is scaled off the RECIPIENT'S income, so a duke does not try to buy a
# king's alliance with 30 gold and an emperor does not overpay a count by 4000.
# `round = yes` on all three: an interaction that pays 47.3184 gold looks broken in
# the tooltip even when the arithmetic is right.
#
# TWO TIERS PER RESOURCE, AND THAT IS THE WHOLE "ONE CORRECTIVE STEP" MECHANISM.
# 30-V2-DESIGN §6.4.3 forbids a convergence loop ("One archetype + at most one
# corrective step; if the package is still negative, abort"). In the INTERACTIONS
# the two tiers are two non-exclusive send_options, so the engine itself searches
# the combinations and — because ai_will_do subtracts for each sweetener — lands on
# the CHEAPEST combination the recipient will accept. In the C-2 PULSE the low tier
# is tried first and the high tier is the single corrective step. Same numbers, same
# fairness, no loop anywhere.
###############################################################################

# --- one-time gold ----------------------------------------------------------
# Low ~= 0.6 of a year of the recipient's income; high ~= 1.8 years.
# Caps: 400 / 1200 gold. The window's own gold stepper runs 0..5000
# (30-V2-DESIGN §3.2), so both tiers sit comfortably inside the range a player can
# offer by hand — an AI must never be able to bid something a player cannot match.
tnt_ai_gold_low_value = {
	value = tnt_ai_recipient_income_value
	multiply = 0.6
	min = 25
	max = 400
	round = yes
}

tnt_ai_gold_high_value = {
	value = tnt_ai_recipient_income_value
	multiply = 1.8
	min = 75
	max = 1200
	round = yes
}

# --- one-time prestige ------------------------------------------------------
# A quarter of what the ACTOR is sitting on, because prestige is his to spend and
# unlike gold it has no income proxy. Clamped 150..2500; the window's prestige
# stepper runs 0..10000 in steps of 50 (30-V2-DESIGN §3.2), so this is a modest
# offer by player standards, which is correct — the AI is not meant to outbid the
# player, only to make a credible one.
# `value = prestige` as a script-value field: REAL,
# GAME\common\script_values\09_mpo_values.txt:2330.
# NOTE the affordability guard is NOT here. It lives on the send_option's is_valid
# and on the apply effect's own limit, because prestige goes NEGATIVE silently
# (30-V2-DESIGN §1 row 2.4) and a value cannot refuse anything.
tnt_ai_prestige_value = {
	value = 0
	if = {
		limit = { exists = scope:actor }
		scope:actor = {
			add = { value = prestige  multiply = 0.25 }
		}
	}
	min = 150
	max = 2500
	round = yes
}


###############################################################################
# PART 3b — THE TWO UNIT PRICES  (2 values, NEW in v3 iteration 2)
#
# The AI-side twins of tnt_gold_per_point_value (tnt_50_values.txt §4.A) and
# tnt_prestige_per_point_value (tnt_53_currency_values.txt §1b), with
# `var:tnt_partner` replaced by `scope:recipient` and `root` by `scope:actor`.
# The full derivation is written out once, in tnt_gold_per_point_value's header;
# the short form is that vanilla's invite-agent bribe menu is the only place in the
# game that states an exchange rate between a resource and an acceptance point:
#     bribe_value gold     -> +10   SSM:552-566   SV:51-90
#     prestige_bribe_value -> +20   SSM:647-666   SV:151-164
#   SSM = GAME\common\scripted_modifiers\00_scheme_scripted_modifiers.txt
#   SV  = GAME\common\script_values\00_scheme_values.txt
#
# NOTE THAT VANILLA'S OWN TWO VALUES READ EXACTLY THESE TWO SCOPES: `bribe_value`
# tests `scope:recipient.primary_title.tier` (SV:63-75) and `prestige_bribe_value`
# adds 25 x `scope:actor.highest_held_title_tier` + 25 x
# `scope:recipient.highest_held_title_tier` (SV:157, :162). So this file's rule —
# never write `root`, read only scope:actor and scope:recipient — is not a
# workaround here, it is the shape vanilla itself uses for the same arithmetic.
#
# There is no piety or influence twin: phase B and phase C-2 price gold, prestige,
# alliance, truce and hook coercion, and nothing else (30-V2-DESIGN §6.3).
###############################################################################

tnt_ai_gold_per_point_value = {
	value = 10
	if = {
		limit = { exists = scope:recipient }
		scope:recipient = {
			if = {
				limit = { highest_held_title_tier >= tier_duchy }
				add = 5
			}
		}
	}
	min = 1
}

tnt_ai_prestige_per_point_value = {
	value = 7.5
	if = {
		limit = { exists = scope:actor }
		scope:actor = {
			add = {
				value = highest_held_title_tier
				multiply = 1.25
			}
		}
	}
	if = {
		limit = { exists = scope:recipient }
		scope:recipient = {
			add = {
				value = highest_held_title_tier
				multiply = 1.25
			}
		}
	}
	min = 10
}


###############################################################################
# PART 4 — THE SAME AMOUNTS, IN ACCEPTANCE POINTS
#
# v3 ITERATION 2 — THE DIVISOR CHANGED, AND IT IS THE SAME CHANGE THE WINDOW MADE.
# These three were `amount / recipient's yearly income x 100`. They are now
# `amount / <that resource's vanilla unit price>` (part 3b), i.e. identical
# arithmetic to tnt_val_gold_p_value and tnt_val_prestige_p_value after their own
# recalibration. The `min = 1` on every divisor is kept: it is what stops a division
# by zero for a landless or freshly-dispossessed recipient.
#
# These are the values the ai_accept `modifier` rows add. They are separate from
# part 3 on purpose: part 3 is what changes hands, part 4 is what it is WORTH, and
# conflating the two is how a mod ends up paying 400 gold for 400 points.
#
# WHAT THE TWO GOLD TIERS NOW BUY, so the archetypes stay legible: gold_low is about
# 0.6 of a year of the recipient's income and gold_high about 1.8, so against a duke
# on 250 gold a year they are 150 and 450 gold, reading +10 and +30. That is a
# credible sweetener and a strong one, on vanilla's own ladder — and it is bounded by
# part 3's caps (400 / 1200 gold, i.e. at most +27 / +80).
###############################################################################

tnt_ai_gold_low_points_value = {
	value = tnt_ai_gold_low_value
	divide = { value = tnt_ai_gold_per_point_value  min = 1 }
}

tnt_ai_gold_high_points_value = {
	value = tnt_ai_gold_high_value
	divide = { value = tnt_ai_gold_per_point_value  min = 1 }
}

# amount / tnt_ai_prestige_per_point_value (part 3b), the same rate the window uses.
#
# THE OLD 0.6 FUDGE IS GONE. It was a hand-averaged stand-in for the window's
# arrogant / humble riders, applied because the gold-relative "1 prestige == 1 gold"
# rate had no tier term of its own. The real rate has one — prestige costs 12.5 per
# point between counts and 20 between emperors — so the fudge is now double
# discounting and it is removed. Personality is still not re-applied here: ai_accept
# carries its own `ai_value_modifier`, and stacking both would price one trait twice.
#
# THE CAP IS NEW AND IT IS LOAD-BEARING. tnt_ai_prestige_value is a quarter of the
# ACTOR'S prestige stock, clamped 150..2500 — a figure with no relation to the
# recipient's tier — so at the top of that range an AI could offer 2500 prestige to a
# poor duke and score +167, deep into vanilla's "forced yes" band (+100..+1000) on a
# single sweetener. Capped at 75, `very_high` on the ladder
# (GAME\common\script_values\00_ai_values.txt:22-34), which is the largest a single
# non-veto row gets anywhere in vanilla's own bribe menu.
tnt_ai_prestige_points_value = {
	value = tnt_ai_prestige_value
	divide = { value = tnt_ai_prestige_per_point_value  min = 1 }
	max = 75
}

# GAME\common\scripted_modifiers\00_marriage_scripted_modifiers.txt:662-670
# (`opinion_modifier = { who = scope:recipient  target = scope:actor
#                        multiplier = 0.75 }`).

# THE SHARED-FAITH BONUS IS GONE ENTIRELY: vanilla has no such row, only penalties

# The three flats below therefore have the same relationship to opinion that

# `"opinion(scope:actor)"` — the QUOTED dotted-call form, mandatory for a
# parenthesised call. REAL: GAME\common\script_values\00_scheme_values.txt:4933
# (`scope:target = { value = "opinion(scope:owner)" }` — entering a scope, then
# reading opinion of another scope, which is exactly the shape below).
# And the NESTED-BLOCK-WITH-A-MULTIPLY form used below is verbatim vanilla,
# GAME\common\script_values\00_scheme_values.txt:2383-2386:
#       add = {
#           value = "opinion(scope:target)"
#           # To bring us roughly in line with skill scaling.
#           multiply = 0.25
#       }
# so the quoted call is legal in a `value =` position inside an arithmetic block,
# not only at the top level of a value. (The mod's own tnt_relmod_opinion_value
# applies its x1.1 at the outer level instead; both forms are proven, and the nested
# one is used here because five further terms follow it in the same value.)
#
# *** THE CULTURE ROW IS THE POST-DEFECT-D1 FORM. *** `has_same_culture_group_as`
# DOES NOT EXIST in 1.19. It is present in
# GAME\common\trigger_localization\00_character_triggers.txt:716 and that file is
# NOT a registry of existing triggers — it retains entries the engine has removed.
# That is how D1 shipped and cost 92 341 error lines in one session
# (50-RUNTIME-LOG-FINDINGS.md §D1, 30-V2-DESIGN §8 R-6, and the mod's own fixed
# copy at tnt_51_relation_values.txt). The two REAL triggers are
#   has_same_culture_as = <char>
#       GAME\common\character_interactions\00_character_interactions.txt:233
#   culture = { has_same_culture_heritage = X.culture }
#       GAME\common\character_interactions\00_education_interactions.txt:1425
#       (negated at :1436) — note the `culture = { }` wrapper: the trigger runs in a
#       CULTURE scope and takes a culture argument.
#
# faith equality is the sanctioned form; `has_same_faith` is blacklisted and does
# not exist. REAL: GAME\common\character_interactions\00_character_interactions.txt:269
# (`faith = scope:actor.faith`).
# faith_hostility_level inside a script value: REAL,
# GAME\common\script_values\00_ep1_script_values.txt:74-80; the constant
# faith_hostile_level = 2 at GAME\common\script_values\02_religion_values.txt:5-12.
# is_close_or_extended_family_of: REAL,
# GAME\common\character_interactions\00_alliance.txt:508, :521.
tnt_ai_relation_value = {
	value = 0
	if = {
		limit = {
			exists = scope:actor
			exists = scope:recipient
		}
		scope:recipient = {
			add = { value = "opinion(scope:actor)"  multiply = 0.75 }

			# No branch for a shared faith: it scores 0, exactly as in vanilla.
			if = {
				limit = {
					faith ?= {
						faith_hostility_level = {
							target = scope:actor.faith
							value >= faith_hostile_level
						}
					}
				}
				add = -35
			}
			else_if = {
				limit = { NOT = { faith = scope:actor.faith } }
				add = -10
			}

			if = {
				limit = { has_same_culture_as = scope:actor }
				add = 5
			}
			else_if = {
				limit = {
					culture ?= {
						NOT = { has_same_culture_heritage = scope:actor.culture }
					}
				}
				add = -10
			}

			if = {
				limit = { is_close_or_extended_family_of = scope:actor }
				add = 10
			}

			if = {
				limit = { is_allied_to = scope:actor }
				add = 10
			}
		}
	}
	round = yes
}

#
# ai_greed / ai_boldness / ai_rationality as script-value fields: REAL,
# GAME\common\script_values\00_activity_values.txt:11, :81, :87; and inside a live
# ai_accept at GAME\common\character_interactions\00_alliance.txt:2034
# (`add = ai_greed`).
# < 0 on the tier edge means the RECIPIENT is the higher tier: a king does not
# haggle with a count, and an emperor still less — hence the two rungs.
# min = 0, not min = 5, and not min = -15: the three AI axes run negative as well as
# positive, but a recipient may not PAY the actor for the privilege of being asked.
tnt_ai_threshold_value = {
	value = 0
	if = {
		limit = { exists = scope:recipient }
		scope:recipient = {
			add = {
				value = ai_greed
				multiply = 0.1
			}
			add = {
				value = ai_boldness
				multiply = 0.05
			}
			subtract = {
				value = ai_rationality
				multiply = 0.05
			}
		}
	}
	if = {
		limit = { tnt_ai_tier_edge_value <= -2 }
		add = 20
	}
	else_if = {
		limit = { tnt_ai_tier_edge_value < 0 }
		add = 10
	}
	min = 0
	max = 40
	round = yes
}

# PART 8 — THE PULSE'S OWN THROTTLE  (PHASE C-2 ONLY)
#
# The per-pulse percentage chance that an AI ruler even LOOKS for a treaty partner
# this year. This one value is the single knob that decides the whole world rate,
# and the full arithmetic is in tnt_72_ai_world.txt's header.
#
# WHY A TIER LADDER. Vanilla's own AI-tier throttle on a yearly pulse,
# GAME\common\on_action\yearly_on_actions.txt:2529-2551 (random_yearly_playable_pulse's
# chance_of_no_event), reads: 30 % skipped for AI kings+, 70 % for AI dukes, 95 %
# for AI counts, with PDX's comments "No need to waste performance here" and "AI
# counts aren't particularly important". The ladder below is that judgement
# inverted into a positive chance: a count is a tenth as likely to broker a treaty
# as a king, which is both cheaper and truer to the period.
#
# WHY has_game_rule LIVES INSIDE A SCRIPT VALUE. Because vanilla does exactly this
# for exactly this purpose. GAME\common\script_values\00_culture_values.txt:448-460
# switches an AI FREQUENCY on
#   has_game_rule = less_common_divergent_culture_ai_frequency / relaxed_… / none_…
# — the direct analogue of our three settings. Also 00_culture_values.txt:255-290,
# 00_character_values.txt:150-162 (random_nickname_frequency_*),
# 06_ce1_epidemics_values.txt:119-143 (epidemic_frequency_*). It is one of the most
# heavily exercised triggers in the whole script_values directory.
#
# THE `off` ROW IS BELT AND BRACES. tnt_72_ai_world.txt's trigger already refuses
# the pulse outright when the rule is off, so this multiply can never be reached in
# practice. It is here so that the value is safe on its own terms: if a future
# author calls it from somewhere new and forgets the gate, the answer is still zero.
#
# THE LADDER WAS DOUBLED 2026-08 (3/7/14 -> 6/14/28), IN THE SAME COMMIT THAT
# DELETED THE PULSE'S SECOND HOST. The old numbers were calibrated for TWO
# evaluations a year (effective chance 1-(1-p)^2, tnt_72_ai_world.txt header
# section 2.4); with one host the per-pulse number IS the annual rate, so the
# rungs double to keep the world rate the ladder was tuned to. That was the old
# header's own standing instruction ("if host 2 is ever deleted ... the ladder
# should be doubled to compensate").
#
# max = 58 caps the whole thing whatever anyone does to the numbers above it.
# 58 and not the old 40: the cap now also does the large-p correction the
# linear doubling cannot. A `common`-rule king lands at 28 x 2.5 = 70, but the
# two-host era's measured effective rate for him was 1-(1-0.35)^2 = 57.75% -
# the cap pins the doubled ladder to THAT ceiling instead of granting a rate
# the mod never actually ran at.
###############################################################################
tnt_ai_world_chance_value = {
	value = 0
	if = {
		limit = { exists = scope:actor }
		scope:actor = {
			if = {
				limit = { highest_held_title_tier >= tier_kingdom }
				add = 28
			}
			else_if = {
				limit = { highest_held_title_tier >= tier_duchy }
				add = 14
			}
			else_if = {
				limit = { highest_held_title_tier >= tier_county }
				add = 6
			}
		}
	}
	if = {
		limit = { has_game_rule = tnt_ai_treaties_common }
		multiply = 2.5
	}
	if = {
		limit = { has_game_rule = tnt_ai_treaties_off }
		multiply = 0
	}
	min = 0
	max = 58
}

# 0.8 IS THE MOD'S OWN NUMBER, AND SINCE 2026-08-25 IT IS LITERALLY THE SAME ONE.
# It was NINE hand-written numerals - this clamp, the four tnt_ai_pay_cap_* and
# the four tnt_player_pay_cap_* - under a header asserting a sharing that did not
# exist. All nine now read tnt_reserve_fraction_value at the foot of this file
# (house law 9). Reading it off the PAYER is the whole difference.
#
# CONSEQUENCE, STATED SO NOBODY REPORTS IT AS A BUG: A MAN WITH AN EMPTY
# TREASURY IS UN-EXTORTABLE, ON BOTH PATHS AND BY THE SAME NUMBER.
# tnt_ai_ultimatum_gold_value floors at 10, so a purse under 13 gold clamps below
# 10. On the LETTER path the dispatch gate in tnt_ai_offer.0001
# (var:tnt_gold_p >= 10) throws it away and A21's entry trigger carries the same
# test as a `root = { gold >= 13 }` row so the draw drops the entry instead of
# burning a lock. On the SANDBOX path E17's affordability row is now that same
# literal 13 on scope:recipient. That is correct - there is nothing to demand.
#
# SECOND CONSEQUENCE, AND IT WILL LOOK LIKE A BUG IN THE LOG THE FIRST TIME IT
# IS SEEN. The points value is UNCAPPED by the user's order, so a crushing
# mismatch produces enormous pressure against a near-empty victim: the run
# carried pts=35,816 and pts=63,030, both asks pinned at the 1800 ceiling. Once
# the clamp binds, those two victims pay 0.8 of whatever they hold and the WDONE
# line reads pts=63030 ask=17. THAT IS THE DESIGN, not a defect - the coin
# ceiling contains the points value and the purse contains the coin. A telemetry
# reader who wants to know whether the clamp bit compares the raw ask against the
# paid one on the same line (`gold=` vs `ask=`, the vocabulary the DEMAND probe
# in tnt_3b_log.txt already uses for this exact mechanic).
#
# SCOPE: scope:recipient is THE PAYER on both paths - the player for A21, the
# victim for E17 - saved by A21's entry body exactly as the sandbox saves it.
# `scope:recipient = { add = { value = gold ... } }` is this file's own shipped
# shape (see tnt_ai_prestige_value above), and the sandbox leg of it is MEASURED:
# tnt_ai_gold_per_point_value reads the recipient's own tier out of that same
# scope block and reproduced all 53 logged asks at its two rates, 10 below duchy
# tier and 15 at duke and above. Reading `gold` there rather than
# `highest_held_title_tier` is the same construct with a different field, and
# `value = gold` plus a multiply inside a script value is vanilla's own
# fifty_percent_of_current_gold (GAME\common\script_values\
# 00_interaction_values.txt:1030-1033) and current_gold_value
# (GAME\common\script_values\00_basic_values.txt:458-461).
###############################################################################
tnt_ai_demand_purse_cap_value = {
	value = 0
	if = {
		limit = { exists = scope:recipient }
		scope:recipient = {
			add = { value = gold  multiply = tnt_reserve_fraction_value }
		}
	}
	min = 0
}

# `max = <named value>` is vanilla's own shape - GAME\common\script_values\
# 00_interaction_values.txt:291 (`max = ransom_cost_value`) and :23. `round = yes`
# for the same reason the ultimatum rounds: this is handed to E1, which pays a
# whole number of gold.
#
# THE NAME IS THE LETTER'S WORD AND THE VALUE IS NOW SHARED - read it as "the
# coin an extortion actually collects". It was NOT renamed to something neutral
# on purpose: the name is read at three shipped sites in two other files and a
# rename buys a word at the price of a citation sweep. What matters is that
# there is ONE of it.
#
# THE PAYMENT GUARD CANNOT REFUSE WHAT THIS HANDS IT, AND THAT PROOF IS OWED
# HERE because E1 (tnt_ai_send_gold_effect) is an `if` with NO `else`: a caller
# whose gate disagrees with E1's guard logs WDONE, moves the goods and pays
# nothing. E1's own header makes "the two guards must name the identical field" a
# rule, and E17's caller gate is now a DIFFERENT ROW from E1's guard - a literal
# 13 against `gold >= tnt_ai_demand_gold_value` - so the rule is satisfied by
# arithmetic instead of by sameness, and here it is:
#   let g = the payer's gold, and the caller has already proved g >= 13.
#   the cap is 0.8g, so this value is round( min(raw, 0.8g) ), raw an integer.
#   if raw <= 0.8g the result is raw, and raw <= 0.8g < g.
#   else the result is round(0.8g) <= 0.8g + 0.5, and 0.8g + 0.5 <= g for every
#   g >= 2.5, which g >= 13 gives with room to spare.
# So the demanded coin is strictly less than the purse for every purse the
# caller lets through, and E1's guard passes by construction. A RAISED BITE b IS
# BOUNDED: bg+0.5 <= g needs g >= 0.5/(1-b), so at the floor 13 b <= 0.96, not 1.
tnt_ai_demand_gold_value = {
	value = tnt_ai_ultimatum_gold_value
	max = tnt_ai_demand_purse_cap_value
	min = 0
	round = yes
}

# CALLING IN A HOOK — the AI's half of V4 item 2   (archetype A6)
#
# The player can spend leverage he already holds (tnt_59_usehook_values.txt); this is
# the same instrument in the sandbox. An AI that holds a hook on a neighbour demands
# gold and pays nothing, and the transfer balances because the debt is real.
#
# THE PRICES ARE THE PLAYER'S OWN, so one calibration serves both: 45 for a favour
# called in, 120 for blackmail-grade leverage. Vanilla's own use_hook decides what
# survives being spent - a weak hook is consumed, a strong one is not - so the archetype
# does not have to model that difference, only price it.
#
# THE DEMAND IS 60% OF THE LEVERAGE, converted at this file's own gold rate, exactly as
# the ultimatum takes 60% of its pressure. Same reason: at 100% the recipient's balance
# lands on zero and every call would hinge on rounding, while 40% of slack pays the
# threshold and any bad blood.
#
# WHY THERE IS NO OPINION PENALTY HERE, unlike the ultimatum. A hook is a debt being
# collected, not a threat being made - vanilla itself charges no opinion for using one,
# and the cost is already paid in the hook: a weak one is gone. Adding a grievance on
# top would make leverage strictly worse than extortion, which is backwards.
###############################################################################
tnt_ai_hookcall_points_value = {
	value = 0
	if = {
		limit = { scope:actor ?= { has_strong_hook = scope:recipient } }
		add = 120
	}
	else_if = {
		limit = { scope:actor ?= { has_weak_hook = scope:recipient } }
		add = 45
	}
	if = {
		limit = {
			tnt_scaled_valuation_enabled_value > 0
			exists = scope:recipient
		}
		multiply = scope:recipient.tnt_scaled_hook_leverage_factor_value
	}
}

tnt_ai_hookcall_gold_value = {
	value = tnt_ai_hookcall_points_value
	multiply = 0.6
	multiply = tnt_ai_gold_per_point_value
	min = 10
	max = 400
	round = yes
}

tnt_ai_hookcall_gold_points_value = {
	value = tnt_ai_hookcall_gold_value
	divide = { value = tnt_ai_gold_per_point_value  min = 1 }
	multiply = -1
}

tnt_ai_deal_hookcall_value = {
	value = 0
	add = tnt_ai_hookcall_points_value
	add = tnt_ai_hookcall_gold_points_value
	add = tnt_ai_relation_value
	add = { value = tnt_ai_threshold_value  multiply = -1 }
	round = yes
}

tnt_ai_title_points_value = {
	value = 0
	if = {
		limit = { exists = scope:tnt_deal_title }
		scope:tnt_deal_title = {
			if = {
				limit = { tnt_scaled_valuation_enabled_value > 0 }
				add = tnt_scaled_county_land_value
			}
			else_if = {
				limit = { tier = tier_barony }
				add = 15
			}
			else_if = {
				limit = { tier = tier_county }
				add = 30
			}
			else_if = {
				limit = { tier = tier_duchy }
				add = 45
			}
			else_if = {
				limit = { tier = tier_kingdom }
				add = 60
			}
			else_if = {
				limit = { tier = tier_empire }
				add = 75
			}
			# THE OUT-OF-DE-JURE RIDER (user item 7b). E20's reach gate now also
			# admits a county that merely BORDERS the buyer's realm; land the
			# buyer holds no de jure claim over is dearer to pry away, and x1.4

			# (tnt_54_multiselect_values.txt, the x1.4 rider on
			# tnt_val_title_multi_r_value) applied in the direction this trade

			# (tnt_ai_deal_title_value), so the round-trip stays exact and the
			# balance still reduces to relation - threshold > 0 - the "land
			# moves only between friendly realms" gate is not loosened by one
			# point. Worst case gold: 30 x 1.4 x 15 = 630, under the 1200 cap.
			# The exists guards keep the two-scope-rule exception declared in
			# this section's header honest: outside E20's chain the whole
			# value is already 0. `exists` on a dotted scope chain: REAL,
			# GAME\common\character_interactions\00_diarch_interactions.txt:417
			# (`exists = scope:actor.primary_title.title_capital_county`).
			if = {
				limit = {
					exists = scope:actor
					exists = scope:actor.primary_title
					NOT = { target_is_de_jure_liege_or_above = scope:actor.primary_title }
				}
				multiply = 1.4
			}
		}
	}

	if = {
		limit = { exists = scope:tnt_deal_artifact }

		}

	}

# THE PAY CAP - HOW DEEP THE AI IS ALLOWED TO REACH INTO ITS OWN PURSE
# V14, USER ITEM 3 (finding F2). Four values, one per currency.
#
# THE PROBLEM THEY SOLVE. The composer's wallet clamps used to read
# `max = scope:tnt_cap_giver.<currency>` - the partner's ENTIRE stock - so every
# letter was priced at the exact edge of the AI's treasury and then, 0 to 200 days
# later, re-validated on that same edge. Two things went wrong at once:
#   1. THE BUG. The Accept option carried `tnt_partner_gold_shortfall_value <= 0`,
#      so the smallest AI outlay between composition and the click flipped the row
#      false and the option refused itself, silently and invisibly - user item 3.
#      That row is now gone and the letter repairs itself instead
#      (tnt_ai_letter_reclamp_effect, tnt_37_ai_offer.txt E-AI-7), but a promise
#      sized at 100% of a purse is a bad promise even when it can be repaired.
#   2. THE IMMERSION. A ruler emptying his treasury to the last coin to buy one
#      courtier reads as absurd. The 2026-08-16 run has a letter offering 22 gold
#      AND 87.36773 prestige - the fraction is the partner's exact stock, i.e.
#      both purses drained to zero, which is how you can spot a bound clamp in
#      the telemetry: an integer amount means the clamp did not bind.
#
# THE FRACTION IS 0.8 AND IT WAS 0.9 FOR ONE COMMIT. V14 user item 3 (F2) set it
# at 0.9 to absorb ordinary AI spending across the letter's life. V14 USER ITEM 2
# (finding I2-B) deepened it to 0.8, and the reconciliation is recorded here
# because this is the only place either number ever appeared:
#   * IT IS VANILLA'S OWN RESERVE FRACTION. eighty_percent_of_max_military_strength,
#     GAME\common\script_values\00_interaction_values.txt:898-903, is the same
#     "keep a fifth back" idea, and it is the value vanilla itself uses to decide
#     whether a proposer is worth listening to (00_marriage_scripted_modifiers.txt
#     :2368). Using the same fifth here means the mod reserves what vanilla
#     reserves.
#   * IT IS WHAT MAKES THE NEW GO/NO-GO ROW BITE. events\tnt_ai_events.txt now
#     refuses a letter unless the proposer gives at least as much as it takes OR
#     the settle actually landed in its band (`tnt_ai_accept_value <= 3`). An AI
#     that can only reach +1 by emptying its treasury now gets 80% of the way
#     instead of 90%, its headline stays further above the ceiling, and the letter
#     is refused instead of shipped. At 0.9 the same letter could still squeak in.
#   * THE COST IS ONE FIFTH OF WHAT A POOR AI CAN OFFER, which is the point: the
#     user's complaint is a count buying a hegemon's daughter with his entire
#     fortune, and that letter should not exist at any rounding.
#

# value keeps Law R7's named-value discipline, matches the shape the rest of this
# file uses, and puts the reserve fraction in ONE place - which is exactly why

# (34.743 / 114.57876 / 127.47671 / 41.01157) under the old 100% clamp.
#
# THE `var:tnt_partner ?= { add = <field> }` SHAPE IS THIS MOD'S OWN, shipped at
# tnt_50_values.txt:289 (`add = yearly_character_income`) and :365
# (`add = current_military_strength`), with the vanilla proof for a scope block
