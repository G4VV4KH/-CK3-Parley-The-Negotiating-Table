# Native-informed currency interests; all public p/r values run on the table owner.
# p = evaluator var:tnt_partner RECEIVES, r = that evaluator SURRENDERS.
# Native stock/capacity helpers run on any character and can be reused by AI-world
# adapters. No persistent variables, resource writes, budget effects or randomness.
#
# Net amount is used ONLY for scoring. The caller must validate BOTH gross legs
# for affordability/permissions before settlement. Netting grants no permission.
#
# For fixed native capacity C, receiving integrates marginal weights 1 below C,
# 1/(1+s) between C and 2C, and 0 above 2C. Surrender integrates marginal costs
# 1 above C, 1+s below C, and 1+3s below C/4. These are differences of continuous
# piecewise-linear potentials. The overlap implementation avoids subtracting two
# large accumulated potentials. Amount splitting telescopes at fixed context.
# Changing government, land, war state or native capacity is a new context; these
# per-resource helpers alone do not claim cross-resource arbitrage prevention.
#
# Existing arrogant/humble/zealous/ambitious and greed/rationality/boldness prices
# remain in the point-conversion layer below and are applied exactly once.
# Capacity bonuses indicate concrete resource use/strategic budget priorities.
#
# Native evidence (installed CK3 1.20.0.4):
# common/script_values/00_ai_values.txt:280 budget numeric reads; :1887 prestige MAA
# common/scripted_triggers/00_ai_value_triggers.txt economic strategy predicates
# common/defines/ai/00_ai.txt:163 desired war chest tracks 18 months maintenance
# common/casus_belli_types/00_religious_war.txt:20 religious spending prerequisites
# common/character_interactions/00_perk_interactions.txt:910 sanctioned loopholes
# common/governments/00_government_types.txt government_has_influence flag
#
# Native capacity is capped at 100000; each income/budget input is bounded before
# arithmetic. Currency stock is read as-is: clipping stock would introduce a
# discontinuity and let successive small transfers evade the clipping threshold.
# Display shares are bounded 0..1, guarded divisors >=0.001, receiving
# factors 0..1, surrender factors 1..7. The documented 1% willingness safety floor
# cannot bind with these bands; no unreported numerical floor is applied.

tnt_interest_gold_native_stock_value = { value = gold }
tnt_interest_prestige_native_stock_value = { value = prestige }
tnt_interest_piety_native_stock_value = { value = piety }
tnt_interest_influence_native_stock_value = { value = this.influence }

tnt_interest_gold_native_capacity_value = {
	value = 100
	desc = tnt_interest_bd_gold_capacity
	add = {
		value = yearly_character_income
		min = 0
		max = 25000
		multiply = 2
		desc = tnt_interest_bd_income_capacity
	}
	# A treasury government funds state reserves with treasury, not personal gold.
	if = {
		limit = { has_treasury = no }
		add = {
			value = war_chest_gold_maximum
			min = 0
			max = 25000
			add = { value = reserved_gold_maximum min = 0 max = 25000 }
			desc = tnt_interest_bd_military_capacity
		}
		if = {
			limit = {
				OR = {
					ai_has_warlike_personality = yes
					ai_has_cautious_personality = yes
					ai_has_conqueror_personality = yes
				}
			}
			add = {
				value = war_chest_gold_maximum
				min = 0
				max = 25000
				desc = tnt_interest_bd_strategy_capacity
			}
		}
		if = {
			limit = { is_at_war = yes }
			add = {
				value = monthly_character_expenses
				min = 0
				max = 2000
				multiply = 12
				desc = tnt_interest_bd_military_capacity
			}
		}
	}
	if = {
		limit = {
			OR = {
				ai_has_economical_boom_personality = yes
				ai_has_pious_builder_personality = yes
				ai_should_focus_on_building_in_their_capital = yes
			}
		}
		add = {
			value = yearly_character_income
			min = 0
			max = 25000
			desc = tnt_interest_bd_strategy_capacity
		}
	}
	min = 100
	max = 100000
}

tnt_interest_piety_native_capacity_value = {
	value = 250
	desc = tnt_interest_bd_resource_capacity
	# General eligibility is a spending opportunity, not a claimed future war.
	if = {
		limit = {
			is_landed = yes
			faith ?= {
				NOR = {
					has_doctrine_parameter = unreformed
					has_doctrine_parameter = holy_wars_forbidden
				}
			}
			NOR = {
				government_has_flag = government_is_landless_adventurer
				government_has_flag = government_is_nomadic
				government_has_flag = government_is_celestial
			}
		}
		add = { value = 500 desc = tnt_interest_bd_religious_capacity }
	}
	if = {
		limit = { has_perk = sanctioned_loopholes_perk }
		add = { value = 500 desc = tnt_interest_bd_religious_capacity }
	}
	min = 100
	max = 100000
}
