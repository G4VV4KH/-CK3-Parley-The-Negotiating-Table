# AI-world negotiation interests. No root or table-owner variable dependency.
# These adapters consume only existing actor/recipient/object scopes. They create
# no shadow table, numeric scope cache, persistent variable or native mutation.
# Active world proposals are synchronous: their tnt_ai_deal_* quote is recomputed
# in the final pre-mutation limit. Off leaves every previous score unchanged.
#
# Gold marginal bands are the same continuous interval integral as tnt61. Amount
# names are fixed by each world archetype; *_gold_* factors run on the evaluating
# character. Native capacity/stock primitives are shared with manual quotations.
# The table and this adapter's single-object advanced factors are parity-tested.
# Territory/income changes are not simulated here; full cross-resource context
# remains a native acceptance limit, not an invented future-plan prediction.
#
# Bilateral voluntary score = min(recipient score, actor score). Each ruler has
# its own relation and bargaining threshold; positive relation is multiplied by
# its receiving factor so an unwanted gift earns no separate friendship payment.
# Pressure is preserved verbatim. Ultimatum/hookcall alter demanded gold cost
# and require the actor to have positive use for receiving it; they do not
# discount threat strength or an already-held hook as an unwanted gift.

tnt_interest_ai_actor_gold_per_point_value = {
	value = 10
	if = { limit = { scope:actor ?= { highest_held_title_tier >= tier_duchy } } add = 5 }
	min = 1
}
tnt_interest_ai_actor_tier_edge_value = {
	value = tnt_ai_tier_edge_value
	multiply = -1
}

tnt_interest_ai_actor_relation_value = {
	value = 0
	if = {
		limit = {
			exists = scope:recipient
			exists = scope:actor
		}
		scope:actor = {
			add = { value = "opinion(scope:recipient)"  multiply = 0.75 }

			# No branch for a shared faith: it scores 0, exactly as in vanilla.
			if = {
				limit = {
					faith ?= {
						faith_hostility_level = {
							target = scope:recipient.faith
							value >= faith_hostile_level
						}
					}
				}
				add = -35
			}
			else_if = {
				limit = { NOT = { faith = scope:recipient.faith } }
				add = -10
			}

			if = {
				limit = { has_same_culture_as = scope:recipient }
				add = 5
			}
			else_if = {
				limit = {
					culture ?= {
						NOT = { has_same_culture_heritage = scope:recipient.culture }
					}
				}
				add = -10
			}

			if = {
				limit = { is_close_or_extended_family_of = scope:recipient }
				add = 10
			}

			if = {
				limit = { is_allied_to = scope:recipient }
				add = 10
			}
		}
	}
	round = yes
}

tnt_interest_ai_actor_threshold_value = {
	value = 0
	if = {
		limit = { exists = scope:actor }
		scope:actor = {
			add = {
				value = ai_greed
				multiply = 0.1
			}
			add = {
				value = ai_boldness
				multiply = 0.05
			}
			subtract = {
				value = ai_rationality
				multiply = 0.05
			}
		}
	}
	if = {
		limit = { tnt_interest_ai_actor_tier_edge_value <= -2 }
		add = 20
	}
	else_if = {
		limit = { tnt_interest_ai_actor_tier_edge_value < 0 }
		add = 10
	}
	min = 0
	max = 40
	round = yes
}
