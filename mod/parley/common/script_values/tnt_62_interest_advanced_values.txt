# Parley negotiation interests: advanced terms, table-owner scope only.
# The evaluator is the current var:tnt_partner. No cached tnt_p/actor/root scope
# is trusted. No effects, game permissions, MCA values, pressure or spent hooks
# are changed here. Base skills, family, de-jure, greed and artifact usability
# remain in the existing valuation layer and are not scored again.
#
# Public API: tnt_interest_<term>_<p|r>_{factor,percent,selected}_value.
# p is incoming benefit factor [0,1]; r is outgoing cost factor [1,100].
# Percent values carry exact signed percentage-point breakdowns.
# Signed contracts additionally expose undirected factor/percent/selected and
# effective_value; a negative base uses the r factor, never a receiving discount.
#
# Native precedents (installed CK3 1.20):
# 00_courtier_and_guest_interactions.txt:189-270, cp:* council vacancies.
# 00_artifact_interactions.txt:498-740, equipped artifacts and personal slots.
# 00_vassal_interactions.txt:77,1140, vassal capacity.
# Existing tnt_54_multiselect_values: domain_size/domain_limit and guarded lists.
# Motive constants are Parley policy, not numbers claimed to come from vanilla.
# Context formulas account for both selected directions. They do not simulate
# all prospective income/contract effects: native whole-deal coupling remains a
# separate acceptance gate. No exact arbitrary-transaction split invariant is
# asserted by these bounded contextual advanced-term factors.

tnt_interest_council_vacancies_value = {
	value = 0
	var:tnt_partner ?= {
		if = { limit = { NOT = { exists = cp:councillor_chancellor } } add = 1 }
		if = { limit = { NOT = { exists = cp:councillor_marshal } } add = 1 }
		if = { limit = { NOT = { exists = cp:councillor_steward } } add = 1 }
		if = { limit = { NOT = { exists = cp:councillor_spymaster } } add = 1 }
	}
}

tnt_interest_domain_incoming_excess_share_value = {
	value = 0
	if = {
		limit = { tnt_domain_titles_p_value > 0 }
		var:tnt_partner ?= { add = domain_size subtract = domain_limit }
		subtract = tnt_domain_titles_r_value
		add = tnt_domain_titles_p_value
		min = 0
		max = tnt_domain_titles_p_value
		divide = tnt_domain_titles_p_value
	}
}
