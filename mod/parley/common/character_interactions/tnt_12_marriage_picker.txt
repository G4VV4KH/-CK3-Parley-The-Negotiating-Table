# PARLEY: THE NEGOTIATING TABLE — THE BORROWED MARRIAGE PANEL.
#
# THE USER: "can we integrate the classic beautiful marriage-arranging panel into
# our deal? So that we choose the line, the grand wedding and so on there, but the
# 'arrange marriage' button is replaced with 'add to deal', and the deal's points
# are counted from there."
#
# THE ANSWER, AND IT IS THE WHOLE POINT OF THIS FILE:
#   special_interaction  and  interface  ARE TWO SEPARATE KEYS.
#     special_interaction = arrange_marriage_interaction   <- THE THING THAT MARRIES
#     interface           = marriage                       <- ONLY the C++ matchmaker panel
# We set the second and we do NOT set the first, so the player gets the vanilla
# picker with NOBODY GETTING MARRIED, and our own on_accept receives the two
# chosen people as real script scopes. Paradox says so in as many words in their
# own shipped schema, GAME\common\character_interactions\_character_interactions.info:576-580:
#
#	Q:	What does the special interaction type arrange_marriage_interaction do?
#	A:	special_interaction = arrange_marriage_interaction will add extra marriage info ...
#		It automatically marries/betroths the secondary participants and sets up
#		alliances and hands out prestige.
#
# and again at .info:376-378, where the marriage is called out by name as a
# HARD-CODED side effect that the special hooks bring with them:
#	# Only executes if the interaction was auto accepted. Done before any other
#	# side effect (E.G., hard coded ones like marriage)
#	pre_auto_accept = effect
#
# Research: _docs\90-MARRIAGE-PANEL-BORROW.md (Route A). Full proof-of-no-marriage
# in the THIS INTERACTION MARRIES NOBODY block further down.
#
# ===========================================================================
# THE TRAP THAT WOULD HAVE KILLED THIS FEATURE SILENTLY
# ===========================================================================
# The C++ panel is PINNED BESIDE THE INTERACTION CONFIRMATION WINDOW. Our front
# door (tnt_10_interaction.txt:300) deliberately suppresses that window with
# needs_confirmation = { always = no } to get its one-click open. THIS
# INTERACTION MUST DO THE OPPOSITE: there is NO needs_confirmation key anywhere
# below, and adding one would delete the very window the panel hangs on. If a
# future author "harmonises" the two interactions by copying that line here, the
# panel stops appearing and on_accept starts staging whatever the scopes happen
# to hold. Do not.
#
# ===========================================================================
# ONE CORRECTION TO THE BRIEF, AND IT IS AN R2 MATTER
# ===========================================================================
# The brief said "set interface = marriage plus the two secondary participant
# keys". The two secondary participant keys DO NOT EXIST ANY MORE and are not
# written below. .info:582-587 documents secondary_actor = marriage /
# secondary_recipient = marriage, but:
#   * grep `^\s*secondary_(actor|recipient)\s*=` over the whole of
#     GAME\common\character_interactions\  -> ZERO matches, and
#   * Paradox's own files carry the syntax COMMENTED OUT — three separate
#     leftovers survive in the 1.19 tree that WGoA2 forked
#     (00_education_interactions.txt:13, :15, :818; 05_bp2_interactions.txt:6,
#     :599; 06_ep3_interactions.txt:4961, all `# secondary_actor = ...`).
# That is a removed key with stale documentation, exactly the class of thing
# Law R3 exists for. THE LIVE MECHANISM IS populate_actor_list /
# populate_recipient_list, which is what vanilla's own arrange_marriage_interaction
# actually uses (00_marriage_interactions.txt:162, :252) and what the one shipping
# mod that already borrows this panel uses
# (mod 3037969445 zz_00_lmfmod_interactions.txt:1304, :1411, with
# `#special_interaction = arrange_marriage_interaction` commented out at :1245).
# Both are written below. Do not add the dead keys.
#
# ===========================================================================
# WHAT THE PANEL DELIVERS, AND WHAT IT DOES NOT
# ===========================================================================
#   scope:secondary_actor      OUR side's spouse       <- the player's household
#   scope:secondary_recipient  THEIR side's spouse     <- the partner's household
#     Vanilla's own comment names them, 00_marriage_interactions.txt:330-331:
#       # actor, recipient ... characters arranging the marriage
#       # secondary_actor, secondary_recipient ... characters that should marry
#     and the direction is confirmed independently by vanilla's cost block at
#     :281, `scope:secondary_actor = { is_close_family_of = scope:actor }`.
#     GETTING THESE TWO BACKWARDS WOULD INVERT EVERY MATCH IN THE DEAL, so the
#     staging effect takes them as named parameters P and R and never guesses.
#   scope:matrilineal          THE LINE                <- send_option flag, below
#   scope:grand_wedding_promise THE GRAND WEDDING      <- send_option flag, below
#
# WHAT IT DOES NOT DELIVER, and our own iteration-4 controls must keep providing:
#   * per-pair line flipping AFTER the pair is committed (tnt_wed_row_line_flip)
#   * which committed pair is the principal        (tnt_wed_set_principal)
#   * removing a committed pair / clearing the deal (tnt_wed_remove, tnt_wed_clear)
#   * the pair list itself, its prices, and every other term in the treaty
#   * the words on the big confirm button. interaction_marriage.gui:229 hardcodes
#     text = "send_marriage_proposal_button_name" ("Send Proposal"). We cannot
#     change it without overriding a vanilla .gui, which Law R1 forbids. What we
#     CAN set is the window HEADER, which is
#     [CharacterInteractionConfirmationWindow.GetSendName] (interaction_marriage.gui:44)
#     and therefore reads our send_name key. So the header says "add to the deal"
#     and the button says "Send Proposal". Localization must not promise otherwise.

tnt_stage_marriage_interaction = {

	# --- identity ------------------------------------------------------------
	category = interaction_category_diplomacy	# .info:18 (Required!)
	icon = icon_marriage				# REAL: 00_marriage_interactions.txt:13, :1199
	interface_priority = 98				# .info:10 — tnt_open_negotiations is 100, the three AI replies are 99
	desc = tnt_stage_marriage_desc			# .info:182

	### THE LINE THAT SITS OVER THE CANDIDATE LIST — the one place in vanilla's window
	### where our text lands NEXT TO its alliance grids.
	###
	### `prompt` is a field of the interaction (_character_interactions.info:195,
	### `prompt = loc_key`) and vanilla uses it in the same spirit at
	### 00_vassal_interactions.txt:21 and 00_diarch_interactions.txt:75. The matchmaker
	### draws it at interaction_marriage.gui:338-345: a centred text_single, above the
	### divider and the candidate list, visible as soon as the list has items - which is
	### exactly where the "Potential alliance" strength bars are drawn per candidate.
	###
	### IT MUST BE ONE SHORT LINE. The widget is a `text_single` with
	### `autoresize = no`, so it does not wrap and a long sentence is simply cut. The
	### full wording - including the advice to use the game's own marriage interaction
	### for an alliance - is carried by the two places that CAN hold a paragraph: the
	### interaction's own desc and the effects block (tnt_stage_no_alliance_tt below).
	prompt = tnt_stage_marriage_prompt
	send_name = tnt_stage_marriage_send		# .info:123 — THIS IS THE PANEL'S HEADER TEXT (interaction_marriage.gui:44)

	# =========================================================================
	# THE ONE LINE THIS WHOLE FILE EXISTS FOR.
	#
	# interface = marriage summons MatchmakerInteractionWindow — the two big
	# portraits, the two candidate lists with all sixteen filters, the marriage-type
	# icon, the matrilineal checkbox and the grand-wedding checkbox. Nothing else.
	# REAL: 00_marriage_interactions.txt:21 (arrange_marriage_interaction) and
	#       :1208 (marry_off_interaction); mod 3037969445 :1244 and :2092.
	#
	# *** THERE IS NO special_interaction KEY IN THIS FILE AND THERE MUST NEVER BE
	# *** ONE. Nor special_ai_interaction. See THIS INTERACTION MARRIES NOBODY.
	# =========================================================================
	interface = marriage

	# --- MANDATORY. Defaults to YES (.info:108). Omit and the interaction
	#     silently vanishes for every ruler outside diplomatic range. Our front
	#     door sets it too (tnt_10_interaction.txt:86) and the two must agree, or
	#     the negotiation opens and the picker refuses to. -----------------------
	use_diplomatic_range = no

	# --- the recipient gets no say. This stages a pair into OUR OWN deal sheet;
	#     the partner's consent is bought later, once, by the whole treaty.
	#     .info:272; 284 exact `auto_accept = yes` lines in vanilla. --------------
	auto_accept = yes

	# --- DELIBERATELY ABSENT KEYS, each for a stated reason -------------------
	# needs_confirmation      would delete the window the panel is pinned to (see
	#                         THE TRAP at the head of this file)
	# special_interaction     would marry people (see THIS INTERACTION MARRIES NOBODY)
	# special_ai_interaction  the AI marriage planner. We are not the AI's business.
	# recipient_recieve_cooldown / ignore_recipient_recieve_cooldown
	#                         vanilla carries these (:33-42) to stop proposal spam.
	#                         WE MUST NOT: a deal with nine marriages in it means
	#                         opening this panel nine times in a row, and a cooldown
	#                         would refuse the second one.
	# ai_targets              without it the engine has no candidate recipient at
	#                         all (.info:490-495), which is one of the three locks
	#                         keeping the AI out. The other two are ai_frequency = 0
	#                         and is_ai = no in is_shown, both below.
	# secondary_actor / secondary_recipient   removed syntax, see the header.
	# cost                    the price of a marriage is the DEAL's price, computed
	#                         by tnt_50_values.txt / tnt_52_marriage_values.txt off
	#                         the committed list. Charging prestige here would
	#                         charge for merely staging a pair.

	# =========================================================================
	# is_shown — TWO JOBS: keep the AI out, and make this openable ONLY from an
	# open negotiation with THE SAME PARTNER.
	#
	# Only scope:actor and scope:recipient exist here (.info:225-233); `root` does
	# not, and the secondary scopes may not exist yet, which is why nothing below
	# mentions them.
	#
	# THE PARTNER LOCK IS THE SECOND ROW PAIR AND IT IS THE SAFETY PROPERTY OF THE
	# WHOLE FEATURE. Without it a player with a negotiation open could right-click
	# any other ruler in the world, pick two people out of this panel and have them
	# staged into a deal with somebody else entirely — a marriage promise made to
	# the wrong man. So: the actor must have a live window (var:tnt_open) and the
	# recipient must BE the partner that window is about (var:tnt_partner).
	# Law R10 satisfied twice: `exists` in front, and `?=` at the dereference. The
	# `var:x ?= { this = <scope> }` read-back is this mod's own shipped idiom
	# (tnt_70_on_actions.txt:174) and is preferred to `var:x ?= <scope>`.
	#
	# THE LAST ROW IS DELIBERATE REUSE, NOT BELT AND BRACES. T26
	# (tnt_show_marriage_p_trigger, tnt_41_gates.txt) is the very trigger that
	# decides whether the marriage row appears in our own window at all: the Tier-4
	# game rule plus both candidate lists being non-empty. Delegating to it means
	# the picker can never be offerable for a negotiation whose marriage row is
	# hidden — ONE gate, two consumers. T26 reads only the current scope and
	# $OTHER$, so entering scope:actor is what makes it correct here.
	# =========================================================================
	is_shown = {
		# CK3 1.20: keep the native effective arranger aligned with the owner
		# of this deal. A puppet/controller mismatch must not borrow the
		# controller's spouse pools or stage variables on a different ruler.
		trigger_if = {
			limit = { exists = scope:puppet_or_actor }
			scope:puppet_or_actor = scope:actor
		}
		scope:actor = {
			is_ai = no			# REAL: 00_alliance.txt:1129, :2103. Lock 1 of 3.
			is_landed = yes			# REAL: 00_character_interactions.txt:26. Also what makes the actor his own matchmaker, which the redirect below relies on.
			exists = var:tnt_open
			exists = var:tnt_partner
			var:tnt_partner ?= { this = scope:recipient }
		}
	}

	# *** V3: THE THREE GATES MOVED OUT OF is_shown, AND THIS IS THE BUG FIX. ***
	#
	# tnt_show_marriage_p_trigger used to sit in is_shown above. That made the
	# whole term silently unavailable: the launch button in our own panel is
	# `enabled = IsPlayerInteractionValid` with
	# `tooltip_when_disabled = GetPlayerInteractionValidBlockers`, and blockers are
	# collected from THIS block - which was empty. So a player whose game rule for
	# advanced terms is off, or who simply has nobody eligible to marry, got a grey
	# button whose tooltip resolved to an empty string. Reported, twice, as "the
	# button is not clickable and does not even hover": the mouse was arriving all
	# along, the button was correctly disabled, and the explanation was blank.
	#
	# is_shown now keeps only "is this our negotiation, with this partner" - the
	# conditions under which the interaction has no business existing at all. Every
	# condition a player could ACT on is here instead, each with its own reason, so
	# the button either works or says why.
	is_valid_showing_failures_only = {
		scope:actor = {
			custom_description = {
				text = tnt_err_advanced_off
				exists = var:tnt_advanced
			}
			custom_description = {
				text = tnt_err_no_wed_cand_p
				has_variable_list = tnt_list_spouse_p
				variable_list_size = {
					name = tnt_list_spouse_p
					value > 0
				}
			}
			custom_description = {
				text = tnt_err_no_wed_cand_r
				has_variable_list = tnt_list_spouse_r
				variable_list_size = {
					name = tnt_list_spouse_r
					value > 0
				}
			}
		}
	}

	# =========================================================================
	# redirect — NORMALISE THE SCOPES. COPIED FROM VANILLA, WITH ONE DELETION.
	#
	# .info:281-289: "Effect changes character interaction scopes - replace any of
	# actor, secondary actor, recipient, secondary recipient and intermediary with
	# any other character". Vanilla's own version is 00_marriage_interactions.txt:94-160
	# and its comment at :137-140 is why this block is not optional:
	#	# marriage interactions are special, and code sets scopes in an unusual way
	#	# secondary recipient must always be set for marriage script,
	#	# but marriage UI only sets primary recipient
	#	# make sure redirect solves this issue
	# i.e. the person the player clicks on the right can arrive as the PRIMARY
	# recipient, and script has to push it down into secondary_recipient itself.
	#
	# BOTH BLOCKS BELOW ARE IDEMPOTENT, AND THAT IS THE POINT. Whether the C++
	# sets the secondary scope directly or hands us the pick as the primary, the
	# same code produces the same end state:
	#   * secondary_recipient already set -> block 1 is guarded off, block 2 finds
	#     a ruler who is his own matchmaker and does nothing. No change.
	#   * recipient = the picked daughter -> block 1 promotes her to
	#     secondary_recipient, block 2 restores recipient = her matchmaker = the
	#     partner. Correct.
	#   * nothing picked yet -> the partner RULER is promoted into
	#     secondary_recipient by block 1, i.e. he arrives pre-selected. The user
	#     reported this and the report was accurate - the panel opens with the
	#     partner already in the right-hand spouse slot instead of empty. CLOSED AS
	#     WON'T-FIX BY HIS OWN EXPLICIT RULING ("Оставить как есть", 2026-08-22):
	#     the only real fix is an out-of-band flag whose failure mode cannot be
	#     tested statically, while the defect is cosmetic - one click on any other
	#     candidate replaces the ruler. V20 tried to close it in-band by negating
	#     block 1's `this = matchmaker` row and that broke picking a ruler at all.
	#     The full record - both purposes of the vanilla block, the V20 regression
	#     and the ruling - is in the box on block 1 below, and it must be kept.
	#     REOPENING THIS REQUIRES A NEW USER RULING, not a tidy-up.
	#
	# WHAT IS DELETED FROM VANILLA'S VERSION, and it must stay deleted:
	#   * the betrothal branches (:107-114, :118-135). They exist so that vanilla's
	#     interaction can FULFIL an existing betrothal. Our candidates are
	#     is_betrothed = no by construction (can_be_picked below, and
	#     tnt_spouse_try_add_effect in tnt_34_marriage.txt) — a pending betrothal is
	#     not ours to trade away — so those branches are dead code here.
	#
	# WHAT IS KEPT AND WHY IT LOOKS DANGEROUS BUT IS NOT: `matchmaker = { save_scope_as = actor }`
	# in block 0 can move scope:actor. Our actor is a LANDED RULER (is_shown), and a
	# landed ruler is his own matchmaker, so `NOT = { this = matchmaker }` is false
	# and block 0 never fires. It is here only for the hypothetical where the C++
	# hands us the LEFT-side pick as the primary actor — then it restores the player
	# and files the pick where it belongs. If it ever fires with a character whose
	# matchmaker is NOT the player, is_shown fails on the next pass and the Send
	# button goes dark. THAT IS THE CORRECT FAILURE: loud and harmless, never a
	# variable written onto a courtier.
	#
	# MARRIAGE_OWNER_IS_MATCHMAKER is the define behind all of this —
	# GAME\common\defines\ai\00_ai.txt:270-275.
	# =========================================================================
	redirect = {
		# --- block 0: OUR SIDE. A no-op for a landed ruler; see above. -------
		scope:actor = {
			if = {
				limit = {
					exists = matchmaker
					NOT = { this = matchmaker }
				}
				save_scope_as = secondary_actor
				matchmaker = {
					save_scope_as = actor
					# 1.20 vanilla redirects both aliases. Keep all deal state
					# on actor; this only normalizes the marriage UI's alias.
					save_scope_as = puppet_or_actor
				}
			}
		}

		# --- THEIR SIDE ------------------------------------------------------
		scope:recipient = {
			# block 1: promote a pick who IS his own matchmaker, i.e. a landed ruler.
			# Vanilla 00_marriage_interactions.txt:141-148, copied positively.
			#
			# *** V20: THIS ROW WAS NEGATED TO KILL THE PRE-SELECTION AND THE NEGATION
			# *** BROKE PICKING A RULER ALTOGETHER. IT IS RESTORED. DO NOT NEGATE IT.
			# *** THE PRE-SELECTION ITSELF IS CLOSED AS WON'T-FIX - USER RULING,
			# *** "Оставить как есть", 2026-08-22. REOPENING IT REQUIRES A NEW USER
			# *** RULING, NOT A TIDY-UP. The analysis below is kept in full because
			# *** it is the map that stops a later round from re-breaking ruler-
			# *** picking the way V20 did while "cleaning up" the default.
			# The behaviour he reported is real - the panel opens with the partner ruler
			# already in the spouse slot and an empty slot would be the better default -
			# but this row is not the place to fix it, because vanilla's block serves
			# TWO purposes at once and the comment at 00_marriage_interactions.txt:137-140
			# says so: "marriage UI only sets primary recipient / make sure redirect
			# solves this issue".
			#   purpose 1  first open: recipient IS the partner ruler and no secondary
			#              exists, so this block promotes him - that is the unwanted default.
			#   purpose 2  the user deliberately picks a ruler out of the grid: the C++
			#              hands the pick back as the PRIMARY recipient (vanilla says so at
			#              :551, "not the secondary characters, because none exist yet"),
			#              and this is the ONLY leg that files him into secondary_recipient.
			# Negating the row makes block 1 and block 2 carry the same condition, so for a
			# self-matchmaker NEITHER fires and scope:secondary_recipient is never written.
			# That scope is the whole picker's handle - this file :483-484, :523-524,
			# :730-731, :750-751 - so the pair cannot be staged and its legality cannot be tested:
			# the partner ruler stops being marriageable at all. A cosmetic wrong default
			# was traded for a functional loss.
			# THE TWO STATES ARE INDISTINGUISHABLE HERE. On first open and on a deliberate
			# pick of that same ruler, scope:recipient is the ruler and secondary_recipient
			# is unset - identical. Any real fix therefore needs a signal from OUTSIDE the
			# redirect (a flag written by the button that opens the panel and cleared on the
			# first redirect pass). THAT DESIGN IS WHY THE THREAD IS CLOSED RATHER THAN
			# FIXED: its failure mode cannot be tested statically - if the clear ever
			# fails to happen, the ruler becomes permanently unpickable, which looks
			# exactly like the V20 regression above - while the defect it would cure is
			# cosmetic, since the pre-selected ruler is replaced by one click on any
			# other candidate. Weighing exactly that trade, the user ruled on
			# 2026-08-22: "Оставить как есть" - it stays as it is. CLOSED AS
			# WON'T-FIX; reopening requires a NEW user ruling, not a tidy-up.
			# The per-row filter stays in can_be_picked below, and T51 door 1
			# (tnt_41_gates.txt:2129) keeps admitting the ruler, as it must.
			if = {
				limit = {
					NOT = { exists = scope:secondary_recipient }
					exists = matchmaker				# REAL: 00_marriage_interactions.txt:143
					this = matchmaker				# REAL: :144
				}
				save_scope_as = secondary_recipient
			}

			# block 2: put the ruler back in the recipient slot. Vanilla :149-158.
			if = {
				limit = {
					exists = matchmaker
					NOT = { this = matchmaker }
				}
				save_scope_as = secondary_recipient
				matchmaker = { save_scope_as = recipient }
			}
		}
	}

	# =========================================================================
	# populate_actor_list / populate_recipient_list — THE TWO CANDIDATE LISTS.
	#
	# .info:291-299: "Everyone sorted into the list 'characters' has the potential
	# of being shown to be selected."
	#
	# *** THESE DO NOT COPY VANILLA'S FOUR-PASS HOUSEHOLD WALK, AND THAT IS THE
	# *** MOST IMPORTANT DECISION IN THIS FILE. ***
	# The mod ALREADY has the answer: tnt_list_spouse_p and tnt_list_spouse_r,
	# built by tnt_fill_spouse_p_effect / tnt_fill_spouse_r_effect
	# (tnt_34_marriage.txt) on every window open, from four passes (the ruler,
	# children, close family, court) filtered by tnt_spouse_try_add_effect —
	# vanilla's marriage_interaction_can_be_picked_trigger plus is_betrothed = no
	# plus the permissive matchmaker rule, de-duplicated. Re-deriving the candidate
	# set here would create a SECOND definition of "who may be offered", and the
	# two would drift: the panel would show a courtier our own picker refuses, or
	# hide one it allows, and the T26/T27 visibility gate (which counts those very
	# lists) would be describing a different panel from the one on screen.
	# So the panel is fed the mod's own lists, and there is exactly ONE candidate
	# rule in the mod.
	#
	# *** BOTH LISTS LIVE ON THE PLAYER — INCLUDING THE PARTNER'S. *** That is the
	# shipped storage model (tnt_34_marriage.txt header, and tnt_fill_title_r_effect
	# at tnt_31_lists.txt:109-130 is the same shape), so BOTH blocks enter
	# scope:actor. Do not "fix" the second one to enter scope:recipient; it would
	# find nothing and the right-hand list would come up empty.
	#
	# has_variable_list guards the read — an absent list is the "Failed to fetch
	# variable" class of error (defect D5, 90 000 log lines).
	# REAL: 00_alliance.txt:3305; 06_ce1_wars.txt:22.
	# `every_in_list = { variable = X  add_to_list = Y }` — REAL, verbatim shape:
	# GAME\common\secret_types\00_secret_types.txt:737-740; also
	# GAME\common\casus_belli_types\10_tgp_china_wars.txt:3073-3076.
	# `add_to_list = characters` is the list name the engine reads — REAL:
	# 00_marriage_interactions.txt:164, :254.
	# =========================================================================
	populate_actor_list = {
		scope:actor = {
			if = {
				limit = { has_variable_list = tnt_list_spouse_p }
				every_in_list = {
					variable = tnt_list_spouse_p
					add_to_list = characters
				}
			}
			# Parley-owned compatibility hook; its frozen body is a no-op
			# (tnt_1f_addon_hooks.txt). MCA 2.0 derives the visible grade from
			# GUI context and ScriptValues, so it neither stamps nor re-orders
			# this list. Keep the call at the tail and keep its SIDE contract:
			# older integrations may know the ABI, while current native order
			# remains untouched. The empty-list case is therefore also a no-op.
			tnt_ma_grade_order_effect = { SIDE = p }
		}
	}

	populate_recipient_list = {
		scope:actor = {
			if = {
				limit = { has_variable_list = tnt_list_spouse_r }
				every_in_list = {
					variable = tnt_list_spouse_r
					add_to_list = characters
				}
			}
			# The r-side twin of the frozen Parley-owned no-op above. MCA 2.0
			# displays a derived value and preserves native candidate order.
			tnt_ma_grade_order_effect = { SIDE = r }
		}
	}

	# =========================================================================
	# can_be_picked — the per-row filter for the FIRST of the two picks.
	#
	# .info:702-706 is precise about what this is and is not:
	#	If we're haven't picked a character already ...
	#		- Trigger: can_be_picked with the current interaction scipe and the
	#		  tested character set as root
	#	Note: can_be_picked is only there to filter out the first character we need
	#	to select and is not part of the Can Send test.
	# So: the CANDIDATE is root here, which is why both rows are unqualified —
	# vanilla's own body is the same single unqualified row (:552-561).
	#
	# It is a second line of defence, not the gate: the lists above were already
	# filtered by the same two triggers when they were built. It is written anyway
	# because the lists are a snapshot taken at window open and a candidate can
	# marry or die while the window sits there.
	# =========================================================================
	can_be_picked = {
		marriage_interaction_can_be_picked_trigger = yes
		is_betrothed = no
	}

	# =========================================================================
