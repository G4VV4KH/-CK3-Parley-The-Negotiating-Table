########################################################################
# E-AI-0 - tnt_ai_offer_recipient_cooldown_effect   root = THE PLAYER
# Called only from the visible trade/demand events' immediate blocks. Failed
# composition must not spend the player's quiet period. The hidden dispatcher
# has no engine floor or provisional recipient reservation; its trigger repeats
# the visible-rate and busy-state guards. A valid draft retains tnt_partner.
# Reuse the same timed variable, so GC/letter cleanup need no new state. A
# refused-envoys decision owns its ten-year lock and must never be shortened.
# Event-immediate effects/variable writes: GAME/events/court_maintenance_events.txt
# lines 318-326. Timed cooldown writes already run in .0001's immediate below.
########################################################################
tnt_ai_offer_recipient_cooldown_effect = {
	if = {
		limit = {
			is_ai = no
			NOT = { exists = var:tnt_envoys_refused }
			NOT = { has_game_rule = tnt_ai_offer_rate_off }
			NOT = { has_game_rule = tnt_ai_offer_rate_test }
		}
		if = {
			limit = { has_game_rule = tnt_ai_offer_rate_frequent }
			set_variable = {
				name = tnt_ai_offer_cooldown
				value = 1
				months = 6
			}
		}
		else_if = {
			limit = { has_game_rule = tnt_ai_offer_rate_rare }
			set_variable = {
				name = tnt_ai_offer_cooldown
				value = 1
				years = 3
			}
		}
		else = {
			set_variable = {
				name = tnt_ai_offer_cooldown
				value = 1
				years = 1
			}
		}
	}
}

# E-AI-1 - tnt_ai_offer_try_effect          root = THE AI PROPOSER
#
# Called from tnt_ai_offer_pulse's `effect` (tnt_71_ai.txt), inside a
# `random = { chance = ... }`.
#
# THE THREE THINGS THIS EFFECT MUST GET RIGHT
#
# 1. IT MUST FIRE THE EVENT FROM INSIDE THE EFFECT, never from the on_action's
#    `events = { }` list. An on_action's effect and its event lists are SEPARATE
#    CHAINS - GAME\common\on_action\_on_actions.info:95, verbatim: "Effects run
#    here create a separate chain than events the on_action fires... Scopes or
#    local variables set in the effect here will not carry over to any event
#    fired by the on_action." The working pattern is `X = { trigger_event = Y }`
#    from inside the effect, which carries the saved scopes AND re-roots the
#    event onto X: GAME\common\on_action\culture_on_actions.txt:63-75 saves
#    scope:hostage and fires bp2_yearly.8050 on the warden, and that event reads
#    scope:hostage throughout with bare fields meaning the warden
#    (GAME\events\dlc\bp2\bp2_yearly.txt:1809-1853).
#
# 2. IT MUST NEVER FIRE ONTO A BUSY PLAYER (30-V2-DESIGN 6.4 constraint 1).
#    Writing the package onto the player is what makes this design cheap, and it
#    is also what would silently destroy a half-built manual offer. The guard is
#    written THREE times on purpose - in the candidate limit here, in the final
#    gate below, and in tnt_ai_offer.0001's own `trigger` - because a queued
#    event is evaluated later than the effect that queued it, and the player can
#    open the window in between.
#
# 3. IT MUST STAY O(1). 43-...md ruling R8: `every_ruler` appears 44 times in
#    vanilla and never once on a periodic on_action; game_start.txt:1363 carries
#    PDX's own "# PUT EVERY_RULER TYPE STUFF HERE FOR PERFORMANCE REASONS". One
#    random_ pick per AI per pulse. Nothing global, no lists.
#
# random_neighboring_top_liege_realm_owner: REAL, 57 call sites, and one of them
# is this exact shape - GAME\common\decisions\00_major_decisions_east_europe.txt:916-919,
# `random_neighboring_top_liege_realm_owner = { limit = { ... } save_scope_as = heretic }`
# followed by a trigger_event. Also
# GAME\common\scripted_effects\00_courtier_guest_management_effects.txt:345 and
# GAME\common\decisions\30_mongol_invasion_decisions.txt:86.
#
# THE LIEGE FALLBACK IS NOT DECORATION. That iterator only ever yields owners of
# NEIGHBOURING TOP-LIEGE REALMS, i.e. independent rulers. A player who is
# somebody's vassal - which is most of the classic starts - would never be
# offered anything at all. Item 1.1 explicitly widened the partner set to include
# your own liege and your own vassals, and hooking or paying your own liege is
# vanilla (GAME\common\hook_types\00_hook_types.txt:201-204
# liege_favour_strong_hook; GAME\events\...\stewardship_duty_events.txt:3074-3078
# `add_hook = { type = favor_hook  target = scope:my_liege }`), so the AI gets
# the same reach through ONE extra link test, taken only when the primary pick
# found nobody. `exists = liege` guards it, and `liege` is a single link rather
# than an iterator, so the cost is one lookup.
#
# WHY THE CHEAP ROWS ARE IN THE LIMIT AND THE REAL GATE RUNS AFTERWARDS: a
# `limit` is a trigger and cannot save a scope, while tnt_partner_valid_trigger
# takes its candidate as a PARAMETER (`CHARACTER = scope:x`, the shipped call
# shape). So the iterator filters on four flat rows - three one-map lookups
# plus user item 26's hatred law, one opinion read - and the winner - a single
# character - is then put through the mod's own partner gate. Cost: one
# scripted trigger, once.
########################################################################
#=======================================================================
# v3 ITERATION 1, USER ITEM 1 - THE TWO COOLDOWNS ARE NOW A GAME RULE
#=======================================================================
# The user never saw a letter. The machinery was fine; the RATE was not. All four
# limits, why they multiply, and what this rule can and cannot reach are set out
# in the header of common\game_rules\tnt_81_ai_rules.txt, which is this author's
# file too. In short: the five settings of tnt_ai_offer_rate set the two
# durations below, and `test` also REMOVES the player-side lock, which is the one
# that blocks every ruler in the world at once.
#
# WHY AN if / else_if LADDER OF LITERALS AND NOT ARITHMETIC: `years = var:x` on
# set_variable has zero vanilla call sites and must not be used
# (00-ARCHITECTURE section 6 item 3). So each setting names its own literal.
#
# THE `off` SETTING IS CHECKED ONCE, AT THE TOP, and wraps the whole body: at
# `off` this effect costs one has_game_rule lookup per AI ruler per pulse and
# does nothing else - no iterator, no scope, no trigger. The pulse host in
# tnt_71_ai.txt is another author's file and needs no change for this.
#
# THE PLAYER-SIDE LOCK IS READ IN THREE PLACES and all three must agree, or the
# world locks itself out at `test` anyway: the two candidate limits and the final
# gate. Each is wrapped in
#     trigger_if = { limit = { NOT = { has_game_rule = tnt_ai_offer_rate_test } } ... }
# so the row simply is not asked at `test`.
#   trigger_if with a has_game_rule limit: REAL,
#     GAME\common\scripted_triggers\00_game_rule_triggers.txt:56-57, :107-108.
# The PROPOSER's own lock is never skipped either, at any setting - but it is no
# longer WRITTEN here: it moved into tnt_ai_offer.0001's immediate, beside
# tnt_log_dispatch_effect. The tombstone below carries the whole argument.
#=======================================================================
tnt_ai_offer_try_effect = {
	if = {
		limit = { NOT = { has_game_rule = tnt_ai_offer_rate_off } }

		# Permanent, not temporary: it has to survive trigger_event into .0001 and
		# from there into .0002. 43-...md 3.A.7 fact 2 - saved scopes propagate
		# through trigger_event.
		save_scope_as = tnt_ai_proposer

		#===============================================================
		# TELEMETRY (v3 iteration 7). Bodies in tnt_3b_log.txt, all behind the
		# tnt_ai_telemetry game rule, which ships OFF - at that setting each call
		# below is one has_game_rule lookup.
		#
		# NOT ONE LINE OF THE LOGIC ABOVE OR BELOW WAS CHANGED to fit this in. The
		# additions are: this call, one `else` on the rule test, two CAND
		# probes, one NOCAND probe, one PICK call and one `else` carrying the
		# rejection-reason ladder. Every original `if`, `limit` and `set_variable`
		# is byte-for-byte what it was.
		#
		# V11, USER ITEM 2: THE BANNER CALL THAT STOOD HERE IS GONE. It fired once
		# per attempt, immediately in front of TRY, under the byte-identical rule
		# gate, and TRY already carries y= - so it was 1:1 duplication. Measured on
		# the 2026-08-16 run: BANNER 2872 = TRY 1108 + WTRY 1764 exactly, i.e. 24.3%
		# of 11,798 telemetry lines and 0% of the information in them. Proof of life
		# is now COUNT(TRY) + COUNT(WTRY) > 0. tnt_log_banner_effect still exists,
		# with no caller; see its tombstone at tnt_3b_log.txt L0.
		#===============================================================
		tnt_log_try_effect = yes

		# ACROSS WATER TOO (V12, user item 2). The plain
		# random_neighboring_top_liege_realm_owner counts only LAND borders, and on
		# an AGOT map that silently deletes a whole continent: a Westerosi player has
		# no land neighbour in Essos, so no Essosi ruler could ever write to him and
		# the pool of INDEPENDENT candidates - the only pool the fealty archetype can
		# draw from - collapsed to the handful of independent realms touching his own.
		# The across-water variant is vanilla's own and is used 25 times in the game
		# files (e.g. GAME\common\scripted_effects\00_ai_conqueror_effects.txt);
		# 67 more sites use its any_ form. Same O(1) pick, same limit rows.
		random_neighboring_and_across_water_top_liege_realm_owner = {
			limit = {
				is_ai = no
				NOT = { exists = var:tnt_partner }
				trigger_if = {
					limit = { NOT = { has_game_rule = tnt_ai_offer_rate_test } }
					NOT = { exists = var:tnt_ai_offer_cooldown }
				}
				# USER ITEM 26(b), the hatred half of the door the player faces
				# (tnt_10_interaction.txt is_valid): never court a target whose
				# opinion of the proposer sits at the -100 clamp. THIS = the
				# candidate, root is absolute (= the proposer). `opinion` inside
				# an iterator limit: REAL,
				# GAME\common\scripted_effects\00_ai_conqueror_effects.txt:485, :501
				# (both inside vassal/courtier iterator limits); the block form is
				# GAME\common\character_interactions\05_bp2_interactions.txt:199-202.
				opinion = {
					target = root
					value >= -99
				}
			}
			save_scope_as = tnt_ai_target
		}

		# TELEMETRY: which of the two searches produced the candidate. This probe
		# MUST stay between the iterator and the liege fallback - after the
		# fallback there is no way to tell the two sources apart.
		if = {
			limit = { exists = scope:tnt_ai_target }
			tnt_log_cand_neigh_effect = yes
		}

		# Second chance, one link, and only if the neighbourhood produced nobody:
		# our own liege. See the header for why this exists.
		if = {
			limit = {
				NOT = { exists = scope:tnt_ai_target }
				exists = liege
			}
			liege = {
				if = {
					limit = {
						is_ai = no
						NOT = { exists = var:tnt_partner }
						trigger_if = {
							limit = { NOT = { has_game_rule = tnt_ai_offer_rate_test } }
							NOT = { exists = var:tnt_ai_offer_cooldown }
						}
						# USER ITEM 26(b) again - the fallback source must obey
						# the same hatred law as the iterator above, or a hated
						# vassal's letter slips through to a liege at the -100
						# clamp (the final gate below re-tests war, not hatred).
						# THIS = the liege, root = the proposing vassal.
						opinion = {
							target = root
							value >= -99
						}
					}
					save_scope_as = tnt_ai_target
					# TELEMETRY. Called with THIS = the liege; root is absolute and
					# stays the proposing AI - the same fact the opinion row above
					# and the final gate's at-war row rely on. RECORDED, NOT FIXED
					# (the effect body is another author's file): the effect fills
					# pro= from THIS, so a CAND|via=liege line names the LIEGE in
					# pro=, not the proposer.
					tnt_log_cand_liege_effect = yes
				}
			}
		}

		#===============================================================
		# THIRD SOURCE (V12): A FELLOW VASSAL OF THE SAME CROWN.
		#
		# WHY IT HAD TO EXIST. The two searches above can only ever produce two
		# kinds of pair: (owner of a neighbouring top-liege realm) x (an
		# INDEPENDENT player), or (any proposer) x (his own liege). So a player
		# who is himself a vassal - which in AGOT's Westeros is most starts -
		# could only ever be written to by his OWN direct vassals. Every other
		# letter in the mod was unreachable for him, and the V12 war-reachability
		# gate then refused the truce archetypes on that one remaining pair,
		# because a liege and his vassal cannot march on each other. The user's
		# own rule names this pair as the legal one - "vassal and vassal of one
		# realm" - and until now no AI path could produce it.
		#
		# WHY random_player AND NOT AN ITERATOR OVER THE REALM. The recipient must
		# be human (is_ai = no on both searches above), so the honest search space
		# is the player list, not the realm tree: random_player is O(number of
		# players), which is ONE in single player, against a top liege's vassal
		# tree which can be hundreds. REAL: GAME\common\on_action\game_start.txt:4597
		# (random_player); its every_ form has 272 vanilla call sites.
		#
		# THE LIMIT IS THE USER'S RULE, WRITTEN OUT. Both sides are vassals, they
		# share a top liege, and neither is above the other - which is exactly
		# what tnt_can_war_each_other_trigger asks. It is not called here because
		# a saved parameter cannot be handed `this` from inside a limit; the rows
		# are its branch 2 verbatim, and the archetypes ask the real trigger again
		# downstream, so the two can never disagree in the permissive direction.
		#
		# ORDER: last. The neighbourhood and the liege keep priority, so nothing
		# about an independent player's letters changes.
		#===============================================================
		if = {
			limit = {
				NOT = { exists = scope:tnt_ai_target }
				is_independent_ruler = no
			}
			random_player = {
				limit = {
					is_ai = no
					NOT = { exists = var:tnt_partner }
					trigger_if = {
						limit = { NOT = { has_game_rule = tnt_ai_offer_rate_test } }
						NOT = { exists = var:tnt_ai_offer_cooldown }
					}
					# USER ITEM 26(b), the same hatred law as both searches above.
					opinion = {
						target = root
						value >= -99
					}
					# The user's rule. THIS = the player, root = the proposer.
					is_landed = yes
					is_independent_ruler = no
					NOR = {
						target_is_liege_or_above = root
						target_is_vassal_or_below = root
					}
					top_liege = { target_is_vassal_or_below = root }
				}
				save_scope_as = tnt_ai_target
			}
			if = {
				limit = { exists = scope:tnt_ai_target }
				tnt_log_cand_realm_effect = yes
			}
		}

		# TELEMETRY: all three searches came up empty. Expected to be the commonest
		# line in the log - the candidate limit requires is_ai = no, so in single
		# player the only legal recipient in the world is the one human, and every
		# AI who is neither his neighbour nor his realm-mate lands here.
		# See tnt_3b_log.txt L3.
		if = {
			limit = { NOT = { exists = scope:tnt_ai_target } }
			tnt_log_nocand_effect = yes
		}

		if = {
			limit = {
				exists = scope:tnt_ai_target
				# root is the AI here, so this is the interaction's own partner
				# gate asked in the normal direction: is that ruler a legal
				# partner for me?
				tnt_partner_valid_trigger = { CHARACTER = scope:tnt_ai_target }
				scope:tnt_ai_target = {
					is_ai = no
					is_landed = yes
					NOT = { exists = var:tnt_open }
					NOT = { exists = var:tnt_partner }
					trigger_if = {
						limit = { NOT = { has_game_rule = tnt_ai_offer_rate_test } }
						NOT = { exists = var:tnt_ai_offer_cooldown }
					}
					# root is absolute inside a re-scoped block, so this reads
					# "not at war with the AI proposer". Since USER ITEM 26 this
					# row is LAW, not just neighbourliness: it is the at-war half
					# of the door the player faces (tnt_10_interaction.txt
					# is_valid). Do not relax it without repealing the door.
					# Item 26's hatred half is NOT re-tested here, deliberately:
					# both candidate sources above already enforce it in the same
					# tick, and this gate is mirrored row-for-row by the
					# rejection ladder in tnt_3b_log.txt (L5, another author's
					# file) - a row added here without its bucket would surface
					# as why=unexplained.
					NOT = { is_at_war_with = root }
				}
			}

			# Hidden drafts do not reserve fifteen days of the player's time.
			# The successful draft owns tnt_partner until its letter is answered;
			# .0002/.0003 alone write the long visible-letter cooldown.
			# PICK counts launches; DISPATCH counts those that still pass the
			# live busy/envoy/rate checks when the hidden event starts.
			tnt_log_pick_effect = yes

			scope:tnt_ai_target = {
				# No recipient write here: an empty draft must leave the next
				# proposer eligible. .0001 rechecks the visible-letter cooldown
				# and existing partner/open guards before changing the table.
				# Re-roots onto the player. The hidden dispatcher composes and
				# then either presents or aborts - see events/tnt_ai_events.txt
				# for why composition cannot happen out here (root would be the
				# AI, and the eight tnt_relmod_* values read `root` explicitly).
				trigger_event = tnt_ai_offer.0001
			}
		}

		#===============================================================
		# TELEMETRY: THE GATE REFUSED A CANDIDATE THAT DID EXIST.
		#
		# The `if` above tests seven things at once, so a single "gate failed"
		# counter would collapse seven different causes into one number - the exact
		# thing the brief forbids. tnt_log_gatefail_effect re-tests the rows one at
		# a time in an if/else_if ladder and names the FIRST one that fails, so
		# exactly one REJECT line is written per refused candidate and the reasons
		# partition the rejections instead of overlapping.
		#
		# The inner `exists` guard keeps this disjoint from the NOCAND probe above:
		# no candidate at all is why=no_candidate, a candidate that was refused is a
		# REJECT with its own reason. Neither can be mistaken for the other.
		#
		# The ladder is itself inside the telemetry rule test, so at the shipped
		# default this whole branch is one has_game_rule lookup.
		#===============================================================
		else = {
			if = {
				limit = { exists = scope:tnt_ai_target }
				tnt_log_gatefail_effect = yes
			}
		}
	}

	# TELEMETRY: the game rule tnt_ai_offer_rate is at `off`, so no AI will ever
	# write to the player and the whole body above was skipped. WITHOUT THIS LINE
	# "the rule is off" and "the machinery is broken" produce identical logs -
	# nothing at all - which is the one outcome the brief rules out.
	#
	# It cannot use scope:tnt_ai_proposer: that save_scope_as is inside the
	# if-branch this else is the sibling of, so the scope does not exist here.
	# tnt_log_rule_off_effect reads THIS instead, and root is the AI.
	else = {
		tnt_log_rule_off_effect = yes
	}
}

# E-AI-2 - tnt_ai_compose_offer_effect      root = THE PLAYER
#
# Requires var:tnt_partner to be set (the AI) and the table to be empty. Called
# from tnt_ai_offer.0001's `immediate`, where root is already the player.
#
# ONE archetype, chosen by random_list. Each entry writes TERMS ONLY; not one of
# them touches gold. The money is settled once, afterwards, by reading the mod's
# own valuation back - so an archetype cannot be mis-priced by construction, and
# a re-tuned value file cannot make the AI look stupid.
#
# Every entry carries its own `trigger`, and entries whose trigger fails are
# excluded from the draw. THERE IS NO LONGER A CATCH-ALL ENTRY, and that is a
# deliberate change: A7 used to have no trigger at all, which guaranteed the draw
# always had one valid entry - for a hegemon, for an imprisoned partner, for an
# existing ally, for a pair already under truce. A7 was the retired tribute, so
# the guarantee it provided was "when nothing else fits, demand tribute", and it
# went with the mechanic.
#
# AN EMPTY DRAW IS THEREFORE REACHABLE AND IS A CORRECT OUTCOME. When no archetype
# qualifies, no tnt_ai_offer_arch is written, the composer writes no terms, and
# tnt_ai_offer.0001's own go/no-go abandons the letter silently. A ruler with
# nothing to propose proposes nothing. The one thing to keep in mind when editing
# this effect: tnt_ai_offer_settle_effect runs OUTSIDE the random_list, so it must
# stay safe on an empty table - it is (every rung is guarded on `exists`).
# (Build 3's note here said a liege-and-vassal pair could never draw A1/A2/A4/A6.
# FOR THE ALLIANCE that is still false and must stay false: T18 carries no realm
# test, and vanilla's negotiate_alliance_interaction has none either - the only
# realm-aware lines in GAME\common\character_interactions\00_alliance.txt are the
# ai_accept MODIFIERS at :2078-2091 and :2100-2114, which price an alliance and
# never forbid one. An alliance is aimed at THIRD parties, so whether the two
# signatories could fight each other is irrelevant to it.
# FOR THE TRUCE entries A1 and A6 it is true again, and by design: V12 user item
# 4 put war-reachability back into T19 through T45
# tnt_can_war_each_other_trigger, so a truce is composed only for a pair that
# could actually fight - two independents, or two vassals of one realm. A1 and A6
# need no row of their own; they inherit it through tnt_can_truce_anyrealm_trigger.)
#
# The archetype index is remembered in tnt_ai_offer_arch so the event desc can
# branch to prose that names exactly the terms this package contains
# (triggered_desc on `var:tnt_ai_offer_arch = N`, a variable-against-literal test
# and never a variable-against-variable one, Law R7).
########################################################################
tnt_ai_compose_offer_effect = {
	# R0.4's prologue, and R13: object-valued variables are read through `?=`.
	var:tnt_partner ?= { save_temporary_scope_as = tnt_ai }

	if = {
		limit = { exists = scope:tnt_ai }

		# TELEMETRY, B2 - THE ONE ROW THAT SEPARATES "NEVER ELIGIBLE" FROM
		# "ELIGIBLE, NEVER DRAWN", which is the diagnosis A19 and A20 lacked for
		# two rounds. Emitted BEFORE the draw, verbose-gated, and the
		# tnt_can_demand_trigger call lives INSIDE the emitter behind the game
		# rule, so at any other telemetry setting this costs one has_game_rule
		# lookup and evaluates no gate at all.
		tnt_log_demandgate_effect = yes

# auto-balance button uses (tnt_39_autobalance.txt section C). That is user ADD-3's
# second half, verbatim: "this can also be used to optimise the AI's deals."
#
#=======================================================================
# WHY THE OLD BODY HAD TO GO, AND IT IS NOT A STYLE ARGUMENT
#=======================================================================
# The two branches this replaces computed their gold from
#     tnt_ai_accept_value / 100 * tnt_partner_year_income_value
# which was correct when one acceptance point was one hundredth of a year of the
# AI's income. v3 ITERATION 2 DELETED THAT CURRENCY. Prices are now flat,
# tier-driven rates lifted straight from vanilla's bribe menu, and the income
# divisor is gone from every valuation - tnt_50_values.txt:272-280 says so in as
# many words ("NEITHER OF THESE TWO FEEDS THE BALANCE ANY MORE"). So the old
# arithmetic converted points into gold at a rate the rest of the mod had stopped
# using, and its two thresholds (12 and 60) were band edges on a scale that had
# shrunk by roughly a factor of ten. An AI settling up at the wrong exchange rate
# is precisely the "mis-priced term destroys value-fairness by construction"
# failure this file's own header warns about.
#
# The replacement reads the SAME four unit prices every valuation reads -
# tnt_gold_per_point_value and its three siblings - so a re-tuned value file moves
# the AI's arithmetic with it and cannot leave the two out of step again. There is
# no exchange rate written in this file any more, which was always the intent.
#
#=======================================================================
# WHAT IT DOES IN BOTH DIRECTIONS
#=======================================================================
#   BELOW MARGIN   the AI would refuse, or sign only by a hair -> it asks the
#                  player to top up, gold first and then prestige, piety and
#                  influence, never beyond what the player actually holds.
#   ABOVE CEILING  the AI is being handed a windfall -> it PAYS FOR IT out of its
#                  own pocket, raising the _r column, never beyond what it holds.
#                  Without this branch a "value-fair" AI would only ever beg.
#   IN BETWEEN     already fair in both directions. Nothing is written, which is
#                  also the cheapest outcome.
# On this path the player's column starts EMPTY - no archetype writes a currency on
# the _p side - so there is nothing for the AI to trim, and "shed the surplus" can
# only mean "pay". That is the one behavioural difference from the player's button
# and tnt_39_autobalance.txt's C2 header sets it out.
#
#=======================================================================
# MARGIN 1 / CEILING 1 - AS CLOSE TO +1 AS THE ARITHMETIC ALLOWS
# (the heading said "CEILING 3" long after the body moved to 1/1; fixed)
#=======================================================================
# THE USER'S REQUIREMENT, TRANSLATED: a mechanism is needed to balance the AI's
# interests to a score as close to +1 as possible. The band is now the
# narrowest one the acceptance test permits: +1 is the smallest passing headline
# (the only test is `tnt_ai_accept_value > 0`, and exactly 0 declines), and +3
# leaves two points of room for the rounding the currency ladder cannot avoid.
#
# THE BAND WAS 8..20 AND WIDTH WAS NOT THE REAL PROBLEM. The step that walked
# toward the band ignored tnt_deal_mult_value, so at opinion +100 it closed twice
# the deficit it aimed at and the AI sailed past any ceiling; at opinion -50 it
# closed half and the letter went out under-priced. A narrow band on top of a
# mis-sized step would have produced worse letters, not better ones - which is why
# tnt_ab_mult_floor_value and the six passes landed first, and the band second.
#
# WHAT THE OLD COMMENT GOT RIGHT, AND HOW IT IS HANDLED NOW. A letter can sit
# unopened for up to 200 days, and .0002's Accept trigger re-tests the verdict when
# it is finally read; the 8-point margin was a buffer against opinion drifting in
# the meantime. A margin of 1 has no such buffer, so the buffer moved to where it
# belongs - the Accept option itself now tolerates drift, on the ground that the AI
# has already given its word (see events/tnt_ai_events.txt, option accept).
# Putting it in the price instead meant every letter was over-priced by 8 points of
# the player's gold to insure against a drift that usually never happened.
#
# THE ABORT IS UNCHANGED AND STILL BELONGS TO .0001. If the top-up cannot be

		# step is sized off the CEILING, not the margin: tnt_ab_surplus_value is

		# rounding on top of that landed at +4 or +5. With ceiling 1 the same machine
		# targets +1 directly. The shape is already shipped and proven by the player's
		# own button, which calls MARGIN = 1 CEILING = 1 (tnt_22_v2.txt), and the
		# post-pay safety sweeten below catches any undershoot to 0.

		tnt_ai_pricebalance_effect = {
			MARGIN = 1
			CEILING = 1
		}

		#===============================================================
		# THE GIFT MARKER (user items 5+9). The balance above CAPS how much
		# of the AI's standing converts into coin per letter
		# (tnt_ai_generosity_cap_value = 25, the ceiling-raise block in
		# tnt_39_autobalance.txt); a letter whose headline STILL exceeds the
		# cap after everything is therefore shipped ON PURPOSE, as a
		# deliberate gift of goodwill - the user's ruling is immersion over
		# arithmetic here, and nothing on this path aborts a too-generous
		# letter.

		# abort in .0001 is headline <= 0, unchanged", WAS FALSE AND WAS MEASURED
		# FALSE. events\tnt_ai_events.txt aborts on FOUR grounds: accept <= 0,
		# tnt_balance_value >= 60, the substance floor, and the fairness OR. On the
		# 2026-08-17 run 19 of 29 letters aborted and only ONE of them was an
		# accept <= 0 case - 17 died on the fairness OR, whose second half was
		# mathematically unsatisfiable for any proposer with a negative relation
		# row. That row is fixed in V16 (Q2-F1: it now reads
		# tnt_ai_unpaid_ask_value <= 9); this comment is corrected so the next
		# author does not re-derive the same wrong picture of .0001.
		#
		# The marker is a plain timed variable so the letter widget can show
		# "sent out of high regard" without a value call: days = 200 matches
		# tnt_ai_offer_arch's own belt, and both letter exits

			limit = { tnt_ai_gift_headroom_value > 0 }

		tnt_log_settle_effect = yes

# E-AI-4 - tnt_ai_offer_clear_effect        root = THE PLAYER
#
# The single exit path. Called by .0001's abort branch, by .0002's Accept,
# Decline and "stop asking" options, by .0002's on_trigger_fail, and by the
# yearly GC in tnt_71_ai.txt.
#
# tnt_close_window_effect does the heavy lifting: it resets the offer, clears the
# seven picker lists and removes tnt_open / tnt_partner / tnt_advanced /
# tnt_infl_ok_p / tnt_infl_ok_r. Its own header (tnt_30_effects.txt E2) states
# the property this reuse depends on - it is "side-effect-free apart from
# variable removal - no toasts, no trigger_event, no scope writes". Calling it
# with no window open is therefore a pure state wipe, and an unguarded
# remove_variable on an absent variable is silent
# (GAME\common\scripted_effects\00_scripted_effects.txt:61-62).
#
# It also removes tnt_result, which tnt_notify_accepted_effect sets to 1 at the
# end of tnt_apply_deal_effect. On this path that is exactly right: there is no
# window to show a banner in, and the out-of-window feedback is the toast that
# has already fired. Same reasoning and same ordering as item 3.5's auto-close in
# events/tnt_events.txt.
#
# The two AI-only variables are removed FIRST, so that a failure inside
# tnt_close_window_effect could not leave them behind.
#
#=======================================================================
# *** THE `NOT = { exists = var:tnt_open }` GUARD IS NOT OPTIONAL ***
#=======================================================================
# A letter can sit on screen for years, and in that time the player may right-
# click a completely different ruler and open a real negotiation. tnt_open is
# then 1, tnt_partner has been overwritten with the NEW recipient and the table
# has been reset (tnt_open_window_effect, E1). If this effect then ran
# unguarded - and it is called from Decline, from "stop asking", from
# on_trigger_fail and from the yearly GC - it would tear down the player's live
# window and wipe the offer they were in the middle of building. That is a
# data-loss bug reachable by clicking Decline on a letter the player left open,
# i.e. by ordinary play.
#
# With the guard, every caller becomes safe at once and nothing is lost by it:
# if a window is open then tnt_open_window_effect has ALREADY overwritten
# everything the AI wrote, so there is nothing left to reclaim. The only two
# variables that could still be stale are the two removed above, and they are
# removed unconditionally.
#
# The GC in tnt_71_ai.txt tests the same condition in its own trigger. That is
# duplication on purpose: the GC's copy documents when it fires, this one
# protects every OTHER caller.
########################################################################
tnt_ai_offer_clear_effect = {
	# TELEMETRY, AND IT MUST BE THE FIRST LINE IN THIS EFFECT. It reports
	# had_arch, which reads var:tnt_ai_offer_arch - the very variable the next line
	# removes. Moving this call below the removes would silently turn every CLEAR
	# line into had_arch=none.
	#
	# The line cannot name WHICH of the six callers fired, because five of them
	# live in files this author does not own. What it is still good for, and how to
	# recover the outcome by joining CLEAR against APPLY, is set out in
	# tnt_3b_log.txt L11.
	tnt_log_clear_effect = yes

	remove_variable = tnt_ai_offer_live
	remove_variable = tnt_ai_offer_arch
	# The goal index travels with the arch index (V10) - same writers, same exits.
	remove_variable = tnt_ai_goal
	# The untimed letter marker (see tnt_ai_offer.0001's immediate): unlike the
	# two timed lines above it never expires on its own, so this remove - reached
	# by every letter exit - is its ONLY removal.
	remove_variable = tnt_ai_letter
	# The gift marker (items 5+9): timed like tnt_ai_offer_arch, removed on
	# every letter exit like it, written by the settle above.
	remove_variable = tnt_ai_gift
	# B2: the demand's stored pressure (archetype A21). Written days = 200 by
	# A21's entry body and read by the refusal rung in tnt_demand_refused_effect,
	# so it travels with the arch and goal indices and leaves with them.
	# NOTE the ONE variable that must NOT be removed here: var:tnt_demand_lock.
	# That is the player's decade-long exposure cap, deliberately outliving the
	# letter it was paid for - removing it on every letter exit would delete the
	# throttle entirely. Its only exits are its own timer and the uninstall sweep
	# (common\decisions\tnt_86_uninstall.txt).
	remove_variable = tnt_demand_pts

	if = {
		limit = { NOT = { exists = var:tnt_open } }
		tnt_close_window_effect = yes
	}
}

# `add_opinion = { target = X  modifier = M }` executed on root does exactly that.
# THE MODIFIER IS REUSED, NEVER DUPLICATED: it is the key the lockout trigger
# reads, and a second modifier of the same meaning would silently break the
# shared 15-year lock in both directions.
# NO UNINTENDED ARITHMETIC FALLOUT, checked: tnt_relmod_opinion_value reads the

	# extortion: GAME\events\lifestyles\subject_extortion_events.txt:43.

		limit = { exists = scope:tnt_deal_partner }
		add_opinion = {
			target = scope:tnt_deal_partner
			modifier = tnt_threat_opinion
		}
	}
}

tnt_demand_refused_effect = {
	#-----------------------------------------------------------------
	# 1. THE PRICE OF LEAVING A CONFRONTATION UNANSWERED - and it is VANILLA'S
	#    OWN GRAMMAR, not an invention. GAME\events\lifestyles\
	#    subject_extortion_events.txt ships FIVE "Back out" options and each one
	#    charges `add_prestige = miniscule_prestige_loss` (:75, :135, :195, :244,
	#    :304); buying a demand off costs `medium_prestige_loss` at :293. The
	#    party who makes a scene and then does nothing pays prestige. On our
	#    letter the man who has a scene made in his hall and neither pays nor
	#    fights is the one who drops it.
	#    THE RUNG WAS MEANT TO BE THE PRESSURE - minor (75 prestige) under 3.75
	#    threat points, medium (150 prestige) at or above - AND THE ACCESS BAR
	#    MAKES IT INERT. RECORDED, NOT
	#    "FIXED": T53 passes only at tnt_threat_points_value >= 100, so a LIVE
	#    var:tnt_demand_pts is never under 5 and MEDIUM ALWAYS FIRES; minor is the
	#    EXPIRY default alone now, which is what the `exists` guard is for (timed
	#    days = 200). A real split would need HIS number. Vanilla's ladder:
	#    GAME\common\script_values\00_basic_values.txt:1000-1005, wrappers :1008-1031.
	#
	# *** THE OUT-GROWN WAIVER, and it is what makes this a game rather than a
	# *** tax: if T53 no longer passes at CLICK time, the prestige is waived
	# *** entirely. Build your host, or outlive his name, and the demand becomes
	# *** a letter from a man who has not looked at a map lately.
	#-----------------------------------------------------------------
	if = {
		limit = {
			exists = scope:tnt_deal_partner
			tnt_can_demand_trigger = { A = scope:tnt_deal_partner  B = root }
		}
		if = {
			limit = {
				exists = var:tnt_demand_pts
				var:tnt_demand_pts >= 3.75
			}
			add_prestige = medium_prestige_loss
		}
		else = {
			add_prestige = minor_prestige_loss
		}
	}

	#-----------------------------------------------------------------
	# 2. AND HIS OWN NAME SUFFERS. A threat that is refused is a threat that has
	#    been measured. minor_dread_loss = -10, the exact mirror of the +10 the
	#    paid branch awards (00_basic_values.txt:726 and :731).
	#    `add_dread = <a *_dread_loss constant>` IS VANILLA'S OWN SHAPE, and its
	#    one call site is the perfect precedent: GAME\events\blackmail_events.txt
	#    :123 charges `add_dread = medium_dread_loss` to a blackmailer whose
	#    threat was faced down. Same mechanic, same direction, vanilla's number.
	#-----------------------------------------------------------------
	scope:tnt_deal_partner ?= { add_dread = minor_dread_loss }

	#-----------------------------------------------------------------
	# 3. THE MEMORY, LAST - see the header: the waiver above must be judged
	#    before this row exists, or T53's tnt_threat_unspent_trigger row would
	#    fail on the modifier this very statement writes.
	#-----------------------------------------------------------------
	if = {
		limit = { exists = scope:tnt_deal_partner }
		add_opinion = {
			target = scope:tnt_deal_partner
			modifier = tnt_threat_opinion
		}
	}
}

#
# WHY THIS EXISTS. The letter's Accept option used to carry four rows testing
# that the PARTNER could still afford what the letter promised:
#     tnt_partner_gold_shortfall_value     <= 0
#     tnt_partner_prestige_shortfall_value <= 0

# the AI's payment at the very edge of its purse (the wallet clamps in

# and the letter can then wait up to 200 days. So the smallest AI outlay flipped
# a row false - and a false trigger row does NOT hide an option in a three-option
# event, it SHOWS it and DISABLES it, looking exactly like a live one
# (GAME\common\defines\graphic\00_graphics.txt:578, EVENT_OPTIONS_SHOWN_HIDE_
# UNAVAILABLE = 4; GAME\gui\shared\event_windows.gui:85 enabled =
# "[EventOption.IsValid]"; the full argument is in the header of
# events\tnt_ai_events.txt). The player was left with a primary answer that
# silently refused the click.
#
# WHAT IT DOES. Instead of refusing, the letter lowers its own promises to what
# the partner actually holds, so the four rows become unfailable and were
# deleted. Called from tnt_ai_offer.0002's accept option inside a hidden_effect,
# BEFORE tnt_apply_deal_effect.
#
# THE THREE INVARIANTS THAT MAKE IT SAFE, and none of them may be relaxed:
#   1. IT ONLY EVER LOWERS. Every write is a max against the holder's stock, so
#      an amount can fall and can never rise. No price on the PLAYER's side is
#      touched, no term is added and no term is created.
#   2. IT ONLY TOUCHES THE _r COLUMN - what the AI pays. The player's _p column
#      is his own consent and is still tested in the option's trigger.
#   3. max BEFORE min, DELIBERATELY - THE ORDER IS LOAD-BEARING, DO NOT TIDY IT.
#      Gold really can go negative in CK3 (vanilla tests it:
#      GAME\common\character_interactions\00_gift.txt:537, `gold < 0`), so a
#      bankrupt partner yields a NEGATIVE ceiling. Clamp operators are applied in
#      the order written - proven by vanilla writing max first on purpose:
#      GAME\common\script_values\00_culture_values.txt:241 (`max = 100` then
#      `min = 0`, this exact shape) and GAME\common\script_values\
#      00_scheme_values.txt:4949 (`max = 50` then `min = -50`, a pair that could
#      not mean anything if the engine reordered them).
#      So max-then-min sends a promise against a bankrupt partner to 0 - the
#      mod's own "no term" value, since every reader gates on `> 0` - whereas
#      min-then-max would leave a NEGATIVE amount on the table.
#      tnt_ab_ai_pay_effect writes min-then-max and is correct only because its
#      limit carries `gold >= 1`; this effect has no such guard by design,
#      because it must cope with exactly the drifted world that guard assumes
#      away. It clamps to the FULL purse, not to tnt_ai_pay_cap_value's reserve
#      (0.9 under F2, deepened to 0.8 by V14 user item 2): the reserve belongs
#      at composition, and shaving a click-time repair by a further fifth would
#      cut deals the partner can perfectly well afford.
#
# RESIDUAL, KNOWN AND ACCEPTED: set_variable does not run in tooltip mode, so on
# a letter where the clamp bites, the accept tooltip still shows the pre-clamp
# figures. An optimistic tooltip beats an unclickable primary answer.
#
# ITS OWN TEMPORARY SCOPE NAME. tnt_39_autobalance.txt:760-762 warns that the
# per-block tnt_cap_giver saves must not be reused across blocks, so this effect
# saves its own tnt_reclamp_giver rather than borrowing that name.
# The `var:x ?= { save_temporary_scope_as = y }` shape is the mod's shipped one
# (tnt_39_autobalance.txt:1898-1900) and satisfies Law R13.
########################################################################
tnt_ai_letter_reclamp_effect = {
	if = {
		limit = { exists = var:tnt_partner }
		var:tnt_partner ?= {
			save_temporary_scope_as = tnt_reclamp_giver
		}

		if = {
			limit = {
				exists = scope:tnt_reclamp_giver
				exists = var:tnt_gold_r
				var:tnt_gold_r > 0
			}
			set_variable = {
				name = tnt_gold_r
				value = {
					value = var:tnt_gold_r
					max = scope:tnt_reclamp_giver.gold
					min = 0
				}
			}
		}

		if = {
			limit = {
				exists = scope:tnt_reclamp_giver
				exists = var:tnt_prestige_r
				var:tnt_prestige_r > 0
			}
			set_variable = {
				name = tnt_prestige_r
				value = {
					value = var:tnt_prestige_r
					max = scope:tnt_reclamp_giver.prestige
					min = 0
				}
			}
		}

		if = {
			limit = {
				exists = scope:tnt_reclamp_giver
				exists = var:tnt_piety_r
				var:tnt_piety_r > 0
			}
			set_variable = {
				name = tnt_piety_r
				value = {
					value = var:tnt_piety_r
					max = scope:tnt_reclamp_giver.piety
					min = 0
				}
			}
		}

		if = {
			limit = {
				exists = scope:tnt_reclamp_giver
				exists = var:tnt_influence_r
				var:tnt_influence_r > 0
			}
			set_variable = {
				name = tnt_influence_r
				value = {
					value = var:tnt_influence_r
					max = scope:tnt_reclamp_giver.influence
					min = 0
				}
			}
		}
	}
}
